//! Instruction set and value representation for the Lotus Modern BASIC bytecode VM.

#[cfg(not(feature = "std"))]
use alloc::{format, string::String};

#[derive(Debug, Clone, PartialEq)]
pub enum Value {
    Number(f64),
    StringLit(String),
    Boolean(bool),
    Nil,
}

impl Value {
    pub fn as_f64(&self) -> Option<f64> {
        match self {
            Value::Number(n) => Some(*n),
            Value::Boolean(b) => Some(if *b { 1.0 } else { 0.0 }),
            Value::StringLit(s) => s.parse::<f64>().ok(),
            Value::Nil => None,
        }
    }

    pub fn is_truthy(&self) -> bool {
        match self {
            Value::Boolean(b) => *b,
            Value::Number(n) => *n != 0.0,
            Value::StringLit(s) => !s.is_empty(),
            Value::Nil => false,
        }
    }

    pub fn to_string_repr(&self) -> String {
        match self {
            Value::Number(n) => format!("{}", n),
            Value::StringLit(s) => s.clone(),
            Value::Boolean(b) => (if *b { "TRUE" } else { "FALSE" }).into(),
            Value::Nil => "NIL".into(),
        }
    }
}

#[derive(Debug, Clone, PartialEq)]
pub enum Opcode {
    Push(Value),
    Load(String),
    Store(String),
    GetCell,
    SetCell,
    Add,
    Sub,
    Mul,
    Div,
    Eq,
    NotEq,
    Lt,
    LtEq,
    Gt,
    GtEq,
    Jump(usize),
    JumpIfFalse(usize),
    Print,
    Halt,
}

//! Single-pass compiler from Modern BASIC source to compact bytecode.

#[cfg(not(feature = "std"))]
use alloc::{string::String, string::ToString, vec::Vec};

use super::bytecode::{Opcode, Value};

#[derive(Debug, PartialEq)]
pub enum CompileError {
    UnexpectedToken(String),
    UnexpectedEof,
    SyntaxError(String),
}

pub struct Compiler<'a> {
    tokens: Vec<Token>,
    pos: usize,
    code: Vec<Opcode>,
    _phantom: core::marker::PhantomData<&'a ()>,
}

#[derive(Debug, Clone, PartialEq)]
enum Token {
    Ident(String),
    Number(f64),
    StringLit(String),
    Plus,
    Minus,
    Star,
    Slash,
    Eq,
    NotEq,
    Lt,
    LtEq,
    Gt,
    GtEq,
    LParen,
    RParen,
    Comma,
    Newline,
}

impl<'a> Compiler<'a> {
    pub fn compile(source: &str) -> Result<Vec<Opcode>, CompileError> {
        let tokens = tokenize_basic(source);
        let mut compiler = Self {
            tokens,
            pos: 0,
            code: Vec::new(),
            _phantom: core::marker::PhantomData,
        };
        compiler.parse_program()?;
        compiler.code.push(Opcode::Halt);
        Ok(compiler.code)
    }

    fn peek(&self) -> Option<&Token> {
        self.tokens.get(self.pos)
    }

    fn advance(&mut self) -> Option<Token> {
        if self.pos < self.tokens.len() {
            let tok = self.tokens[self.pos].clone();
            self.pos += 1;
            Some(tok)
        } else {
            None
        }
    }

    fn skip_newlines(&mut self) {
        while let Some(Token::Newline) = self.peek() {
            self.advance();
        }
    }

    fn parse_program(&mut self) -> Result<(), CompileError> {
        self.skip_newlines();
        while self.pos < self.tokens.len() {
            self.parse_statement()?;
            self.skip_newlines();
        }
        Ok(())
    }

    fn parse_statement(&mut self) -> Result<(), CompileError> {
        self.skip_newlines();
        let tok = match self.peek() {
            Some(t) => t.clone(),
            None => return Ok(()),
        };

        if let Token::Ident(word) = tok {
            let upper = word.to_ascii_uppercase();
            match upper.as_str() {
                "SUB" => {
                    self.advance(); // consume SUB
                    // consume name and optional ()
                    if let Some(Token::Ident(_)) = self.peek() {
                        self.advance();
                    }
                    if self.peek() == Some(&Token::LParen) {
                        self.advance();
                        if self.peek() == Some(&Token::RParen) {
                            self.advance();
                        }
                    }
                    // Parse until END SUB
                    while let Some(t) = self.peek() {
                        if let Token::Ident(w) = t {
                            if w.to_ascii_uppercase() == "END" {
                                self.advance();
                                if let Some(Token::Ident(s)) = self.peek() {
                                    if s.to_ascii_uppercase() == "SUB" {
                                        self.advance();
                                        break;
                                    }
                                }
                            }
                        }
                        self.parse_statement()?;
                        self.skip_newlines();
                    }
                    Ok(())
                }
                "LET" => {
                    self.advance(); // consume LET
                    let var_name = match self.advance() {
                        Some(Token::Ident(name)) => name,
                        _ => return Err(CompileError::SyntaxError("Expected variable name after LET".into())),
                    };
                    match self.advance() {
                        Some(Token::Eq) => {}
                        _ => return Err(CompileError::SyntaxError("Expected '=' after variable name in LET".into())),
                    }
                    self.parse_expr()?;
                    self.code.push(Opcode::Store(var_name));
                    Ok(())
                }
                "SET_CELL" => {
                    self.advance(); // consume SET_CELL
                    match self.advance() {
                        Some(Token::LParen) => {}
                        _ => return Err(CompileError::SyntaxError("Expected '(' after SET_CELL".into())),
                    }
                    // Arg 1: Target (coord or name)
                    self.parse_expr()?;
                    match self.advance() {
                        Some(Token::Comma) => {}
                        _ => return Err(CompileError::SyntaxError("Expected ',' between SET_CELL arguments".into())),
                    }
                    // Arg 2: Value
                    self.parse_expr()?;
                    match self.advance() {
                        Some(Token::RParen) => {}
                        _ => return Err(CompileError::SyntaxError("Expected ')' after SET_CELL arguments".into())),
                    }
                    self.code.push(Opcode::SetCell);
                    Ok(())
                }
                "PRINT" => {
                    self.advance(); // consume PRINT
                    self.parse_expr()?;
                    self.code.push(Opcode::Print);
                    Ok(())
                }
                "IF" => {
                    self.advance(); // consume IF
                    self.parse_expr()?;
                    if let Some(Token::Ident(w)) = self.peek() {
                        if w.to_ascii_uppercase() == "THEN" {
                            self.advance();
                        }
                    }

                    let jump_false_idx = self.code.len();
                    self.code.push(Opcode::JumpIfFalse(0)); // Placeholder

                    // Body
                    self.skip_newlines();
                    while let Some(t) = self.peek() {
                        if let Token::Ident(w) = t {
                            let u = w.to_ascii_uppercase();
                            if u == "END" || u == "ELSE" {
                                break;
                            }
                        }
                        self.parse_statement()?;
                        self.skip_newlines();
                    }

                    // Check for ELSE
                    let mut has_else = false;
                    let mut jump_end_idx = 0;
                    if let Some(Token::Ident(w)) = self.peek() {
                        if w.to_ascii_uppercase() == "ELSE" {
                            self.advance(); // consume ELSE
                            has_else = true;
                            jump_end_idx = self.code.len();
                            self.code.push(Opcode::Jump(0)); // Placeholder to jump over ELSE body
                            // Patch jump_false to point here
                            self.code[jump_false_idx] = Opcode::JumpIfFalse(self.code.len());

                            self.skip_newlines();
                            while let Some(t) = self.peek() {
                                if let Token::Ident(w) = t {
                                    if w.to_ascii_uppercase() == "END" {
                                        break;
                                    }
                                }
                                self.parse_statement()?;
                                self.skip_newlines();
                            }
                        }
                    }

                    // Expect END IF
                    if let Some(Token::Ident(w)) = self.advance() {
                        if w.to_ascii_uppercase() == "END" {
                            if let Some(Token::Ident(sub)) = self.advance() {
                                if sub.to_ascii_uppercase() != "IF" {
                                    return Err(CompileError::SyntaxError("Expected 'END IF'".into()));
                                }
                            }
                        }
                    }

                    if has_else {
                        self.code[jump_end_idx] = Opcode::Jump(self.code.len());
                    } else {
                        self.code[jump_false_idx] = Opcode::JumpIfFalse(self.code.len());
                    }

                    Ok(())
                }
                "WHILE" => {
                    self.advance(); // consume WHILE
                    let loop_start = self.code.len();
                    self.parse_expr()?;
                    let jump_false_idx = self.code.len();
                    self.code.push(Opcode::JumpIfFalse(0)); // Placeholder

                    self.skip_newlines();
                    while let Some(t) = self.peek() {
                        if let Token::Ident(w) = t {
                            if w.to_ascii_uppercase() == "WEND" {
                                self.advance(); // consume WEND
                                break;
                            }
                        }
                        self.parse_statement()?;
                        self.skip_newlines();
                    }

                    // Jump back to start
                    self.code.push(Opcode::Jump(loop_start));
                    self.code[jump_false_idx] = Opcode::JumpIfFalse(self.code.len());
                    Ok(())
                }
                _ => {
                    // Implicit LET: var = expr
                    self.advance(); // consume var name
                    if self.peek() == Some(&Token::Eq) {
                        self.advance(); // consume '='
                        self.parse_expr()?;
                        self.code.push(Opcode::Store(word));
                        Ok(())
                    } else {
                        Err(CompileError::SyntaxError(word))
                    }
                }
            }
        } else {
            self.advance();
            Ok(())
        }
    }

    fn parse_expr(&mut self) -> Result<(), CompileError> {
        self.parse_comparison()
    }

    fn parse_comparison(&mut self) -> Result<(), CompileError> {
        self.parse_addition()?;
        while let Some(tok) = self.peek() {
            let op = match tok {
                Token::Eq => Opcode::Eq,
                Token::NotEq => Opcode::NotEq,
                Token::Lt => Opcode::Lt,
                Token::LtEq => Opcode::LtEq,
                Token::Gt => Opcode::Gt,
                Token::GtEq => Opcode::GtEq,
                _ => break,
            };
            self.advance();
            self.parse_addition()?;
            self.code.push(op);
        }
        Ok(())
    }

    fn parse_addition(&mut self) -> Result<(), CompileError> {
        self.parse_multiplication()?;
        while let Some(tok) = self.peek() {
            let op = match tok {
                Token::Plus => Opcode::Add,
                Token::Minus => Opcode::Sub,
                _ => break,
            };
            self.advance();
            self.parse_multiplication()?;
            self.code.push(op);
        }
        Ok(())
    }

    fn parse_multiplication(&mut self) -> Result<(), CompileError> {
        self.parse_primary()?;
        while let Some(tok) = self.peek() {
            let op = match tok {
                Token::Star => Opcode::Mul,
                Token::Slash => Opcode::Div,
                _ => break,
            };
            self.advance();
            self.parse_primary()?;
            self.code.push(op);
        }
        Ok(())
    }

    fn parse_primary(&mut self) -> Result<(), CompileError> {
        let tok = self.advance().ok_or(CompileError::UnexpectedEof)?;
        match tok {
            Token::Number(n) => {
                self.code.push(Opcode::Push(Value::Number(n)));
                Ok(())
            }
            Token::StringLit(s) => {
                self.code.push(Opcode::Push(Value::StringLit(s)));
                Ok(())
            }
            Token::LParen => {
                self.parse_expr()?;
                match self.advance() {
                    Some(Token::RParen) => Ok(()),
                    _ => Err(CompileError::SyntaxError("Expected ')'".into())),
                }
            }
            Token::Ident(name) => {
                let upper = name.to_ascii_uppercase();
                if upper == "GET_CELL" {
                    match self.advance() {
                        Some(Token::LParen) => {}
                        _ => return Err(CompileError::SyntaxError("Expected '(' after GET_CELL".into())),
                    }
                    self.parse_expr()?;
                    match self.advance() {
                        Some(Token::RParen) => {}
                        _ => return Err(CompileError::SyntaxError("Expected ')' after GET_CELL argument".into())),
                    }
                    self.code.push(Opcode::GetCell);
                    Ok(())
                } else if upper == "TRUE" {
                    self.code.push(Opcode::Push(Value::Boolean(true)));
                    Ok(())
                } else if upper == "FALSE" {
                    self.code.push(Opcode::Push(Value::Boolean(false)));
                    Ok(())
                } else {
                    self.code.push(Opcode::Load(name));
                    Ok(())
                }
            }
            _ => Err(CompileError::UnexpectedToken("Unexpected primary expression".into())),
        }
    }
}

fn tokenize_basic(source: &str) -> Vec<Token> {
    let mut tokens = Vec::new();
    let chars: Vec<char> = source.chars().collect();
    let mut i = 0;

    while i < chars.len() {
        let c = chars[i];
        if c.is_ascii_whitespace() {
            if c == '\n' {
                tokens.push(Token::Newline);
            }
            i += 1;
            continue;
        }

        match c {
            '+' => tokens.push(Token::Plus),
            '-' => tokens.push(Token::Minus),
            '*' => tokens.push(Token::Star),
            '/' => tokens.push(Token::Slash),
            '(' => tokens.push(Token::LParen),
            ')' => tokens.push(Token::RParen),
            ',' => tokens.push(Token::Comma),
            '=' => tokens.push(Token::Eq),
            '<' => {
                if i + 1 < chars.len() && chars[i + 1] == '>' {
                    tokens.push(Token::NotEq);
                    i += 1;
                } else if i + 1 < chars.len() && chars[i + 1] == '=' {
                    tokens.push(Token::LtEq);
                    i += 1;
                } else {
                    tokens.push(Token::Lt);
                }
            }
            '>' => {
                if i + 1 < chars.len() && chars[i + 1] == '=' {
                    tokens.push(Token::GtEq);
                    i += 1;
                } else {
                    tokens.push(Token::Gt);
                }
            }
            '"' => {
                i += 1;
                let mut s = String::new();
                while i < chars.len() && chars[i] != '"' {
                    s.push(chars[i]);
                    i += 1;
                }
                tokens.push(Token::StringLit(s));
            }
            ch if ch.is_ascii_digit() || ch == '.' => {
                let mut num_str = String::from(ch);
                while i + 1 < chars.len() && (chars[i + 1].is_ascii_digit() || chars[i + 1] == '.') {
                    i += 1;
                    num_str.push(chars[i]);
                }
                let val = num_str.parse::<f64>().unwrap_or(0.0);
                tokens.push(Token::Number(val));
            }
            ch if ch.is_alphabetic() || ch == '_' => {
                let mut ident = String::from(ch);
                while i + 1 < chars.len() && (chars[i + 1].is_alphanumeric() || chars[i + 1] == '_') {
                    i += 1;
                    ident.push(chars[i]);
                }
                tokens.push(Token::Ident(ident));
            }
            _ => {}
        }
        i += 1;
    }

    tokens
}

        let script = r#"

        "#;

        let writes = BasicRunner::execute(&mut sheet, Coord::new(0, 0), script, &[], &[], None).unwrap();
        assert_eq!(writes, 1);

        // Verify A1 was updated to 300.0
        match &sheet.get_cell(Coord::new(0, 0)).unwrap().value {
            CellValue::Number(n) => assert_eq!(*n, 300.0),

        }

        // Verify dependent formula A2 automatically recalculated to 450.0 (300 * 1.5)!
        match &sheet.get_cell(Coord::new(0, 1)).unwrap().value {

//! Transactional, cell-scoped execution coordinator for embedded Modern BASIC.
//!
//! Runtime access is limited to workbook-resident cells. Host-side import and
//! export operations remain outside the runtime; scripts cannot access the
//! filesystem, network, processes, or other operating-system services.

#[cfg(not(feature = "std"))]
use alloc::{format, string::String};

use super::{
    compiler::{CompileError, Compiler},
    vm::{Vm, VmError},
};
use crate::{
    calc::CalcEngine,
    model::{Coord, Sheet},
};

#[derive(Debug)]
pub enum BasicError {
    Compile(CompileError),
    Runtime(VmError),
    Recalc(String),
}

pub struct BasicRunner;

impl BasicRunner {
    /// Compiles and executes a Modern BASIC script against a worksheet.
    ///
    /// Transactional semantics:
    /// - The owning cell is readable and writable by default.
    /// - Other cells require explicit read and/or write grants.
    /// - All writes are staged in a sandbox.
    /// - If the script completes successfully within its grants and step limits,
    ///   the staged writes commit atomically, and all affected formulas recalculate.
    /// - If an error or budget violation occurs, staged writes are discarded and the sheet is unmodified.
    pub fn execute(
        sheet: &mut Sheet,
        owner: Coord,
        source: &str,
        readable_grants: &[Coord],
        writable_grants: &[Coord],
        max_steps: Option<usize>,
    ) -> Result<usize, BasicError> {
        let code = Compiler::compile(source).map_err(BasicError::Compile)?;

        let mut vm = Vm::new(sheet, &code).with_cell_access(owner, readable_grants, writable_grants);
        if let Some(steps) = max_steps {
            vm = vm.with_step_limit(steps);
        }

        vm.run().map_err(BasicError::Runtime)?;

        // Atomic Commit of staged writes
        let num_writes = vm.staged_writes.len();
        for (coord, val) in vm.staged_writes {
            if let Some(cell) = sheet.cells.get_mut(&coord) {
                cell.value = val;
            } else {
                sheet.set_cell(crate::model::Cell::new(coord, val));
            }
        }

        // Trigger DAG recalculation
        CalcEngine::recalculate_sheet(sheet, None).map_err(BasicError::Recalc)?;

        Ok(num_writes)
    }
}

//! Virtual Machine with cell confinement and transactional write staging.

#[cfg(not(feature = "std"))]
use alloc::{collections::{BTreeMap, BTreeSet}, format, string::String, vec::Vec};
#[cfg(feature = "std")]
use std::collections::{BTreeMap, BTreeSet};

use super::bytecode::{Opcode, Value};
use crate::model::{CellValue, Coord, Sheet};

#[derive(Debug, PartialEq)]
pub enum VmError {
    StepBudgetExceeded,
    CellAccessDenied { coord: Coord, operation: CellOperation },
    WriteToFormulaCell(Coord),
    StackUnderflow,
    TypeMismatch(&'static str),
    DivisionByZero,
    InvalidCellTarget(String),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellOperation {
    Read,
    Write,
}

pub struct Vm<'a> {
    sheet: &'a Sheet,
    code: &'a [Opcode],
    ip: usize,
    stack: Vec<Value>,
    vars: BTreeMap<String, Value>,
    steps: usize,
    max_steps: usize,
    pub staged_writes: BTreeMap<Coord, CellValue>,
    readable_cells: BTreeSet<Coord>,
    writable_cells: BTreeSet<Coord>,
}

impl<'a> Vm<'a> {
    pub fn new(sheet: &'a Sheet, code: &'a [Opcode]) -> Self {
        Self {
            sheet,
            code,
            ip: 0,
            stack: Vec::new(),
            vars: BTreeMap::new(),
            steps: 0,
            max_steps: 100_000,
            staged_writes: BTreeMap::new(),
            readable_cells: BTreeSet::new(),
            writable_cells: BTreeSet::new(),
        }
    }

    pub fn with_step_limit(mut self, max_steps: usize) -> Self {
        self.max_steps = max_steps;
        self
    }

    /// Grant this procedure access to its own cell and explicitly listed cells.
    pub fn with_cell_access(mut self, owner: Coord, readable: &[Coord], writable: &[Coord]) -> Self {
        self.readable_cells.insert(owner);
        self.writable_cells.insert(owner);
        self.readable_cells.extend(readable.iter().copied());
        self.writable_cells.extend(writable.iter().copied());
        self
    }

    pub fn run(&mut self) -> Result<usize, VmError> {
        while self.ip < self.code.len() {
            self.steps += 1;
            if self.steps > self.max_steps {
                return Err(VmError::StepBudgetExceeded);
            }

            match &self.code[self.ip] {
                Opcode::Push(val) => {
                    self.stack.push(val.clone());
                    self.ip += 1;
                }
                Opcode::Load(name) => {
                    let val = self.vars.get(name).cloned().unwrap_or(Value::Nil);
                    self.stack.push(val);
                    self.ip += 1;
                }
                Opcode::Store(name) => {
                    let val = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    self.vars.insert(name.clone(), val);
                    self.ip += 1;
                }
                Opcode::GetCell => {
                    let target = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let coord = self.resolve_coord(&target)?;
                    if !self.readable_cells.contains(&coord) {
                        return Err(VmError::CellAccessDenied { coord, operation: CellOperation::Read });
                    }
                    let val = self.read_cell_value(coord);
                    self.stack.push(val);
                    self.ip += 1;
                }
                Opcode::SetCell => {
                    let val = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let target = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let coord = self.resolve_coord(&target)?;

                    if !self.writable_cells.contains(&coord) {
                        return Err(VmError::CellAccessDenied { coord, operation: CellOperation::Write });
                    }
                    if matches!(self.sheet.get_cell(coord).map(|cell| &cell.value), Some(CellValue::Formula { .. })) {
                        return Err(VmError::WriteToFormulaCell(coord));
                    }

                    // Convert Value to CellValue and stage it
                    let cell_val = match val {
                        Value::Number(n) => CellValue::Number(n),
                        Value::StringLit(s) => CellValue::Text(s),
                        Value::Boolean(b) => CellValue::Boolean(b),
                        Value::Nil => CellValue::Empty,
                    };
                    self.staged_writes.insert(coord, cell_val);
                    self.ip += 1;
                }
                Opcode::Add => {
                    let r = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let l = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    match (l.as_f64(), r.as_f64()) {
                        (Some(a), Some(b)) => self.stack.push(Value::Number(a + b)),
                        _ => self.stack.push(Value::StringLit(format!("{}{}", l.to_string_repr(), r.to_string_repr()))),
                    }
                    self.ip += 1;
                }
                Opcode::Sub => {
                    let r = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let l = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    match (l.as_f64(), r.as_f64()) {
                        (Some(a), Some(b)) => self.stack.push(Value::Number(a - b)),
                        _ => return Err(VmError::TypeMismatch("Expected numbers for '-'")),
                    }
                    self.ip += 1;
                }
                Opcode::Mul => {
                    let r = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let l = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    match (l.as_f64(), r.as_f64()) {
                        (Some(a), Some(b)) => self.stack.push(Value::Number(a * b)),
                        _ => return Err(VmError::TypeMismatch("Expected numbers for '*'")),
                    }
                    self.ip += 1;
                }
                Opcode::Div => {
                    let r = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let l = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    match (l.as_f64(), r.as_f64()) {
                        (Some(_), Some(b)) if b == 0.0 => return Err(VmError::DivisionByZero),
                        (Some(a), Some(b)) => self.stack.push(Value::Number(a / b)),
                        _ => return Err(VmError::TypeMismatch("Expected numbers for '/'")),
                    }
                    self.ip += 1;
                }
                Opcode::Eq => {
                    let r = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let l = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    self.stack.push(Value::Boolean(l == r));
                    self.ip += 1;
                }
                Opcode::NotEq => {
                    let r = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let l = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    self.stack.push(Value::Boolean(l != r));
                    self.ip += 1;
                }
                Opcode::Lt => {
                    let r = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let l = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    match (l.as_f64(), r.as_f64()) {
                        (Some(a), Some(b)) => self.stack.push(Value::Boolean(a < b)),
                        _ => self.stack.push(Value::Boolean(l.to_string_repr() < r.to_string_repr())),
                    }
                    self.ip += 1;
                }
                Opcode::LtEq => {
                    let r = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let l = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    match (l.as_f64(), r.as_f64()) {
                        (Some(a), Some(b)) => self.stack.push(Value::Boolean(a <= b)),
                        _ => self.stack.push(Value::Boolean(l.to_string_repr() <= r.to_string_repr())),
                    }
                    self.ip += 1;
                }
                Opcode::Gt => {
                    let r = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let l = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    match (l.as_f64(), r.as_f64()) {
                        (Some(a), Some(b)) => self.stack.push(Value::Boolean(a > b)),
                        _ => self.stack.push(Value::Boolean(l.to_string_repr() > r.to_string_repr())),
                    }
                    self.ip += 1;
                }
                Opcode::GtEq => {
                    let r = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    let l = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    match (l.as_f64(), r.as_f64()) {
                        (Some(a), Some(b)) => self.stack.push(Value::Boolean(a >= b)),
                        _ => self.stack.push(Value::Boolean(l.to_string_repr() >= r.to_string_repr())),
                    }
                    self.ip += 1;
                }
                Opcode::Jump(target) => {
                    self.ip = *target;
                }
                Opcode::JumpIfFalse(target) => {
                    let cond = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    if !cond.is_truthy() {
                        self.ip = *target;
                    } else {
                        self.ip += 1;
                    }
                }
                Opcode::Print => {
                    let _ = self.stack.pop().ok_or(VmError::StackUnderflow)?;
                    self.ip += 1;
                }
                Opcode::Halt => break,
            }
        }

        Ok(self.staged_writes.len())
    }

    fn resolve_coord(&self, val: &Value) -> Result<Coord, VmError> {
        match val {
            Value::StringLit(s) => {
                if let Some(c) = Coord::parse_a1(s) {
                    return Ok(c);
                }
                // Try resolving named cell
                for cell in self.sheet.cells.values() {
                    if cell.name.as_deref() == Some(s) {
                        return Ok(cell.coord);
                    }
                }
                Err(VmError::InvalidCellTarget(s.clone()))
            }
            _ => Err(VmError::InvalidCellTarget(val.to_string_repr())),
        }
    }

    fn read_cell_value(&self, coord: Coord) -> Value {
        // First check locally staged writes
        if let Some(staged) = self.staged_writes.get(&coord) {
            return match staged {
                CellValue::Number(n) => Value::Number(*n),
                CellValue::Text(t) => Value::StringLit(t.clone()),
                CellValue::Boolean(b) => Value::Boolean(*b),
                _ => Value::Nil,
            };
        }

        // Read from sheet
        match self.sheet.get_cell(coord) {
            Some(cell) => match &cell.value {
                CellValue::Number(n) => Value::Number(*n),
                CellValue::Text(t) => Value::StringLit(t.clone()),
                CellValue::Boolean(b) => Value::Boolean(*b),
                CellValue::Formula { cached_num, cached_text, cached_bool, .. } => {
                    if let Some(n) = cached_num {
                        Value::Number(*n)
                    } else if let Some(value) = cached_bool {
                        Value::Boolean(*value)
                    } else if let Some(t) = cached_text {
                        Value::StringLit(t.clone())
                    } else {
                        Value::Nil
                    }
                }
                _ => Value::Nil,
            },
            None => Value::Nil,
        }
    }
}

//! Dependency Graph (DAG) and Incremental Topological Recalculator.

#[cfg(not(feature = "std"))]
use alloc::{
    collections::{BTreeMap, BTreeSet},
    string::String,
    vec::Vec,
};
#[cfg(feature = "std")]
use std::collections::{BTreeMap, BTreeSet};

use super::{
    ast::Expr,
    eval::{EvalValue, Evaluator},
    parser::Parser,
};
use crate::model::{CellValue, Coord, Sheet, Workbook};

pub struct CalcEngine;

impl CalcEngine {
    /// Recalculates all formulas across a sheet using topological dependency ordering.
    /// Detects circular references and marks them with `#CYCLE!`.
    pub fn recalculate_sheet(sheet: &mut Sheet, workbook: Option<&Workbook>) -> Result<usize, String> {
        // 1. Parse all formula cells and extract dependencies
        let mut parsed_formulas: BTreeMap<Coord, (Expr, BTreeSet<Coord>)> = BTreeMap::new();

        for cell in sheet.cells.values() {
            if let CellValue::Formula { expr, .. } = &cell.value {
                if let Ok(ast) = Parser::parse_str(expr) {
                    let mut deps = BTreeSet::new();
                    collect_dependencies(&ast, &mut deps, sheet);
                    parsed_formulas.insert(cell.coord, (ast, deps));
                }
            }
        }

        let formula_cells: BTreeSet<Coord> = parsed_formulas.keys().copied().collect();
        let mut dependents: BTreeMap<Coord, BTreeSet<Coord>> = BTreeMap::new();
        let mut in_degree: BTreeMap<Coord, usize> = BTreeMap::new();

        for (coord, (_, deps)) in &parsed_formulas {
            let mut formula_deps_count = 0;
            for dep in deps {
                if formula_cells.contains(dep) {
                    dependents.entry(*dep).or_default().insert(*coord);
                    formula_deps_count += 1;
                }
            }
            in_degree.insert(*coord, formula_deps_count);
        }

        // 2. Cycle Detection using Kahn's algorithm / Topological sort
        let mut queue: Vec<Coord> = Vec::new();
        let mut remaining_in_degree = in_degree.clone();

        for (coord, deg) in &in_degree {
            if *deg == 0 {
                queue.push(*coord);
            }
        }

        let mut eval_order: Vec<Coord> = Vec::new();
        while let Some(coord) = queue.pop() {
            eval_order.push(coord);

            if let Some(downstream) = dependents.get(&coord) {
                for next in downstream {
                    if let Some(deg) = remaining_in_degree.get_mut(next) {
                        *deg = deg.saturating_sub(1);
                        if *deg == 0 {
                            queue.push(*next);
                        }
                    }
                }
            }
        }

        // If eval_order does not contain all formulas, cycles exist!
        let mut cycle_cells = BTreeSet::new();
        for (coord, deg) in &remaining_in_degree {
            if *deg > 0 {
                cycle_cells.insert(*coord);
            }
        }

        // Mark cycle cells with #CYCLE!
        for coord in &cycle_cells {
            if let Some(cell) = sheet.cells.get_mut(coord) {
                if let CellValue::Formula { ref mut cached_num, ref mut cached_text, ref mut cached_bool, .. } = cell.value {
                    *cached_num = None;
                    *cached_text = Some("#CYCLE!".into());
                    *cached_bool = None;
                }
            }
        }

        // 3. Evaluate in topological order
        let mut count = 0;
        for coord in eval_order {
            if let Some((ast, _)) = parsed_formulas.get(&coord) {
                let eval = Evaluator::new(sheet, workbook);
                let result = eval.eval(ast);

                if let Some(cell) = sheet.cells.get_mut(&coord) {
                    if let CellValue::Formula { ref mut cached_num, ref mut cached_text, ref mut cached_bool, .. } = cell.value {
                        match result {
                            EvalValue::Number(n) => {
                                *cached_num = Some(n);
                                *cached_text = None;
                                *cached_bool = None;
                            }
                            EvalValue::Text(t) => {
                                *cached_num = None;
                                *cached_text = Some(t);
                                *cached_bool = None;
                            }
                            EvalValue::Boolean(b) => {
                                *cached_num = None;
                                *cached_text = None;
                                *cached_bool = Some(b);
                            }
                            EvalValue::Empty => {
                                *cached_num = None;
                                *cached_text = None;
                                *cached_bool = None;
                            }
                            EvalValue::Error(e) => {
                                *cached_num = None;
                                *cached_text = Some(e);
                                *cached_bool = None;
                            }
                        }
                    }
                }
                count += 1;
            }
        }

        Ok(count)
    }

    /// Recalculates all sheets in the entire workbook.
    pub fn recalculate_workbook(wb: &mut Workbook) -> Result<usize, String> {
        let mut total = 0;
        let num_sheets = wb.sheets.len();
        for i in 0..num_sheets {
            let mut sheet = wb.sheets[i].clone();
            let count = Self::recalculate_sheet(&mut sheet, None)?;
            wb.sheets[i] = sheet;
            total += count;
        }
        Ok(total)
    }
}

fn collect_dependencies(expr: &Expr, deps: &mut BTreeSet<Coord>, sheet: &Sheet) {
    match expr {
        Expr::CellRef(c) => {
            deps.insert(*c);
        }
        Expr::RangeRef { top_left, bottom_right } => {
            for col in top_left.col..=bottom_right.col {
                for row in top_left.row..=bottom_right.row {
                    deps.insert(Coord::new(col, row));
                }
            }
        }
        Expr::NamedRef(name) => {
            if let Some(cell) = sheet.cells.values().find(|cell| cell.name.as_deref() == Some(name)) {
                deps.insert(cell.coord);
            }
        }
        Expr::Unary { expr, .. } => collect_dependencies(expr, deps, sheet),
        Expr::Binary { left, right, .. } => {
            collect_dependencies(left, deps, sheet);
            collect_dependencies(right, deps, sheet);
        }
        Expr::FunctionCall { args, .. } => {
            for arg in args {
                collect_dependencies(arg, deps, sheet);
            }
        }
        _ => {}
    }
}

//! Evaluator and built-in function engine for spreadsheet formulas.

#[cfg(not(feature = "std"))]
use alloc::{format, string::String, vec::Vec};

use super::ast::{BinaryOp, Expr, UnaryOp};
use crate::model::{CellValue, Coord, Sheet, Workbook};

#[derive(Debug, Clone, PartialEq)]
pub enum EvalValue {
    Number(f64),
    Text(String),
    Boolean(bool),
    Empty,
    Error(String),
}

impl EvalValue {
    pub fn as_f64(&self) -> Option<f64> {
        match self {
            EvalValue::Number(n) => Some(*n),
            EvalValue::Boolean(b) => Some(if *b { 1.0 } else { 0.0 }),
            EvalValue::Empty => Some(0.0),
            EvalValue::Text(t) => t.parse::<f64>().ok(),
            EvalValue::Error(_) => None,
        }
    }

    pub fn as_string(&self) -> String {
        match self {
            EvalValue::Number(n) => format!("{}", n),
            EvalValue::Text(t) => t.clone(),
            EvalValue::Boolean(b) => (if *b { "TRUE" } else { "FALSE" }).into(),
            EvalValue::Empty => String::new(),
            EvalValue::Error(e) => e.clone(),
        }
    }

    pub fn is_truthy(&self) -> bool {
        match self {
            EvalValue::Boolean(b) => *b,
            EvalValue::Number(n) => *n != 0.0,
            EvalValue::Text(t) => !t.is_empty(),
            EvalValue::Empty => false,
            EvalValue::Error(_) => false,
        }
    }
}

pub struct Evaluator<'a> {
    pub sheet: &'a Sheet,
    pub workbook: Option<&'a Workbook>,
}

impl<'a> Evaluator<'a> {
    pub fn new(sheet: &'a Sheet, workbook: Option<&'a Workbook>) -> Self {
        Self { sheet, workbook }
    }

    pub fn eval(&self, expr: &Expr) -> EvalValue {
        match expr {
            Expr::Number(n) => EvalValue::Number(*n),
            Expr::Text(t) => EvalValue::Text(t.clone()),
            Expr::Boolean(b) => EvalValue::Boolean(*b),

            Expr::CellRef(coord) => self.resolve_cell(*coord),

            Expr::NamedRef(name) => self.resolve_named(name),

            Expr::RangeRef { .. } => {
                // Standalone range reference without aggregation returns #VALUE!
                EvalValue::Error("#VALUE!".into())
            }

            Expr::Unary { op, expr } => {
                let val = self.eval(expr);
                match op {
                    UnaryOp::Neg => match val.as_f64() {
                        Some(n) => EvalValue::Number(-n),
                        None => EvalValue::Error("#VALUE!".into()),
                    },
                    UnaryOp::Not => EvalValue::Boolean(!val.is_truthy()),
                }
            }

            Expr::Binary { op, left, right } => {
                let l = self.eval(left);
                let r = self.eval(right);
                self.eval_binary(*op, l, r)
            }

            Expr::FunctionCall { name, args } => self.eval_function(name, args),
        }
    }

    fn resolve_cell(&self, coord: Coord) -> EvalValue {
        match self.sheet.get_cell(coord) {
            Some(cell) => match &cell.value {
                CellValue::Number(n) => EvalValue::Number(*n),
                CellValue::Text(t) => EvalValue::Text(t.clone()),
                CellValue::Boolean(b) => EvalValue::Boolean(*b),
                CellValue::Formula { cached_num, cached_text, cached_bool, .. } => {
                    if let Some(n) = cached_num {
                        EvalValue::Number(*n)
                    } else if let Some(value) = cached_bool {
                        EvalValue::Boolean(*value)
                    } else if let Some(t) = cached_text {
                        EvalValue::Text(t.clone())
                    } else {
                        EvalValue::Empty
                    }
                }
                CellValue::Empty => EvalValue::Empty,
                CellValue::Error(e) => EvalValue::Error(e.clone()),
                CellValue::Basic { .. } | CellValue::Blob { .. } => EvalValue::Error("#VALUE!".into()),
            },
            None => EvalValue::Empty,
        }
    }

    fn resolve_named(&self, name: &str) -> EvalValue {
        // First check current sheet
        for cell in self.sheet.cells.values() {
            if cell.name.as_deref() == Some(name) {
                return self.resolve_cell(cell.coord);
            }
        }

        // Then check full workbook if available
        if let Some(wb) = self.workbook {
            for sheet in &wb.sheets {
                for cell in sheet.cells.values() {
                    if cell.name.as_deref() == Some(name) {
                        let eval = Evaluator::new(sheet, Some(wb));
                        return eval.resolve_cell(cell.coord);
                    }
                }
            }
        }

        EvalValue::Error("#NAME?".into())
    }

    fn eval_binary(&self, op: BinaryOp, l: EvalValue, r: EvalValue) -> EvalValue {
        match op {
            BinaryOp::Add => match (l.as_f64(), r.as_f64()) {
                (Some(a), Some(b)) => EvalValue::Number(a + b),
                _ => EvalValue::Error("#VALUE!".into()),
            },
            BinaryOp::Sub => match (l.as_f64(), r.as_f64()) {
                (Some(a), Some(b)) => EvalValue::Number(a - b),
                _ => EvalValue::Error("#VALUE!".into()),
            },
            BinaryOp::Mul => match (l.as_f64(), r.as_f64()) {
                (Some(a), Some(b)) => EvalValue::Number(a * b),
                _ => EvalValue::Error("#VALUE!".into()),
            },
            BinaryOp::Div => match (l.as_f64(), r.as_f64()) {
                (Some(_), Some(b)) if b == 0.0 => EvalValue::Error("#DIV/0!".into()),
                (Some(a), Some(b)) => EvalValue::Number(a / b),
                _ => EvalValue::Error("#VALUE!".into()),
            },
            BinaryOp::Mod => match (l.as_f64(), r.as_f64()) {
                (Some(_), Some(b)) if b == 0.0 => EvalValue::Error("#DIV/0!".into()),
                (Some(a), Some(b)) => EvalValue::Number(a % b),
                _ => EvalValue::Error("#VALUE!".into()),
            },
            BinaryOp::Pow => match (l.as_f64(), r.as_f64()) {
                (Some(a), Some(b)) => {
                    #[cfg(feature = "std")]
                    return EvalValue::Number(a.powf(b));
                    #[cfg(not(feature = "std"))]
                    return EvalValue::Number(libm::pow(a, b));
                }
                _ => EvalValue::Error("#VALUE!".into()),
            },
            BinaryOp::Concat => EvalValue::Text(format!("{}{}", l.as_string(), r.as_string())),
            BinaryOp::Eq => EvalValue::Boolean(l == r),
            BinaryOp::NotEq => EvalValue::Boolean(l != r),
            BinaryOp::Lt => match (l.as_f64(), r.as_f64()) {
                (Some(a), Some(b)) => EvalValue::Boolean(a < b),
                _ => EvalValue::Boolean(l.as_string() < r.as_string()),
            },
            BinaryOp::LtEq => match (l.as_f64(), r.as_f64()) {
                (Some(a), Some(b)) => EvalValue::Boolean(a <= b),
                _ => EvalValue::Boolean(l.as_string() <= r.as_string()),
            },
            BinaryOp::Gt => match (l.as_f64(), r.as_f64()) {
                (Some(a), Some(b)) => EvalValue::Boolean(a > b),
                _ => EvalValue::Boolean(l.as_string() > r.as_string()),
            },
            BinaryOp::GtEq => match (l.as_f64(), r.as_f64()) {
                (Some(a), Some(b)) => EvalValue::Boolean(a >= b),
                _ => EvalValue::Boolean(l.as_string() >= r.as_string()),
            },
        }
    }

    fn collect_numbers_from_args(&self, args: &[Expr]) -> Vec<f64> {
        let mut numbers = Vec::new();
        for arg in args {
            match arg {
                Expr::RangeRef { top_left, bottom_right } => {
                    for col in top_left.col..=bottom_right.col {
                        for row in top_left.row..=bottom_right.row {
                            let val = self.resolve_cell(Coord::new(col, row));
                            if let Some(n) = val.as_f64() {
                                numbers.push(n);
                            }
                        }
                    }
                }
                _ => {
                    let val = self.eval(arg);
                    if let Some(n) = val.as_f64() {
                        numbers.push(n);
                    }
                }
            }
        }
        numbers
    }

            // =============================

                        Some(_) => EvalValue::Error("#NUM!".into()),

                } else {

                }

                        Some(n) if n > 0.0 => EvalValue::Number(1.0),
                        Some(n) if n < 0.0 => EvalValue::Number(-1.0),

                    let start_idx = start.saturating_sub(1);
                    let result: String = s.chars().skip(start_idx).take(count).collect();

                    result.push_str(&self.eval(arg).as_string());

                        if c == ' ' {
                            if !prev_space {
                                result.push(' ');
                            }
                            prev_space = true;
                        } else {

                            prev_space = false;

                    if result.ends_with(' ') {

                    EvalValue::Text(result)
                } else {

                }

                } else {

                    let count = self.eval(&args[1]).as_f64().unwrap_or(0.0) as usize;
                    EvalValue::Text(s.repeat(count))

                if args.len() >= 3 {
                    let text = self.eval(&args[0]).as_string();
                    let old = self.eval(&args[1]).as_string();
                    let new = self.eval(&args[2]).as_string();
                    if let Some(instance_arg) = args.get(3) {
                        let instance = self.eval(instance_arg).as_f64().unwrap_or(1.0) as usize;
                        // Replace only the Nth occurrence
                        let mut result = String::new();
                        let mut count = 0usize;
                        let mut remaining = text.as_str();
                        while let Some(pos) = remaining.find(&old) {
                            count += 1;
                            if count == instance {
                                result.push_str(&remaining[..pos]);
                                result.push_str(&new);
                                result.push_str(&remaining[pos + old.len()..]);
                                return EvalValue::Text(result);
                            } else {
                                result.push_str(&remaining[..pos + old.len()]);
                                remaining = &remaining[pos + old.len()..];
                            }
                        }
                        result.push_str(remaining);
                        EvalValue::Text(result)
                    } else {
                        // Replace all occurrences
                        EvalValue::Text(text.replace(&old, &new))

                    EvalValue::Error("#VALUE!".into())
                }
            }

                    let s = self.eval(arg).as_string();
                    match s.trim().parse::<f64>() {

                        Err(_) => EvalValue::Error("#VALUE!".into()),

                    EvalValue::Error("#VALUE!".into())

            _ => EvalValue::Error(format!("#UNKNOWN_FN:{}!", name)),

    /// Helper for single-argument math functions (SIN, COS, etc.)
    fn eval_single_math(&self, args: &[Expr], f: fn(f64) -> f64) -> EvalValue {
        if let Some(arg) = args.first() {
            match self.eval(arg).as_f64() {
                Some(n) => EvalValue::Number(f(n)),
                None => EvalValue::Error("#VALUE!".into()),
            }
        } else {
            EvalValue::Error("#VALUE!".into())
        }
    }

    /// VLOOKUP(lookup_value, range, col_index, [range_lookup])
    fn eval_vlookup(&self, args: &[Expr]) -> EvalValue {
        if args.len() < 3 {
            return EvalValue::Error("#VALUE!".into());
        }
        let lookup = self.eval(&args[0]);
        let (top_left, bottom_right) = match &args[1] {
            Expr::RangeRef { top_left, bottom_right } => (*top_left, *bottom_right),
            _ => return EvalValue::Error("#VALUE!".into()),
        };
        let col_index = match self.eval(&args[2]).as_f64() {
            Some(n) => n as usize,
            None => return EvalValue::Error("#VALUE!".into()),
        };
        if col_index < 1 || col_index > (bottom_right.col - top_left.col + 1) as usize {
            return EvalValue::Error("#REF!".into());
        }
        let exact = if let Some(arg) = args.get(3) {
            !self.eval(arg).is_truthy() // FALSE = exact match
        } else {
            false // default is approximate (TRUE)
        };

        // Search down the first column
        for row in top_left.row..=bottom_right.row {
            let cell_val = self.resolve_cell(Coord::new(top_left.col, row));
            let found = if exact {
                cell_val == lookup
            } else {
                // Approximate: find last value <= lookup_value (assumes sorted)
                match (cell_val.as_f64(), lookup.as_f64()) {
                    (Some(a), Some(b)) => a <= b,
                    _ => cell_val.as_string() <= lookup.as_string(),
                }
            };
            if exact && found {
                let result_col = top_left.col + (col_index as u32 - 1);
                return self.resolve_cell(Coord::new(result_col, row));
            }
            if !exact && !found {
                // We've gone past; use previous row
                if row > top_left.row {
                    let result_col = top_left.col + (col_index as u32 - 1);
                    return self.resolve_cell(Coord::new(result_col, row - 1));
                } else {
                    return EvalValue::Error("#N/A".into());
                }
            }
        }
        if exact {
            EvalValue::Error("#N/A".into())
        } else {
            // Largest value in range
            let result_col = top_left.col + (col_index as u32 - 1);
            self.resolve_cell(Coord::new(result_col, bottom_right.row))
        }
    }

    /// HLOOKUP(lookup_value, range, row_index, [range_lookup])
    fn eval_hlookup(&self, args: &[Expr]) -> EvalValue {

    /// INDEX(range, row_num, [col_num])
    fn eval_index(&self, args: &[Expr]) -> EvalValue {
        if args.is_empty() {
            return EvalValue::Error("#VALUE!".into());
        }
        let (top_left, _bottom_right) = match &args[0] {
            Expr::RangeRef { top_left, bottom_right } => (*top_left, *bottom_right),
            _ => return EvalValue::Error("#VALUE!".into()),
        };
        let row_num = if args.len() > 1 {
            self.eval(&args[1]).as_f64().unwrap_or(1.0) as u32
        } else {
            1
        };
        let col_num = if args.len() > 2 {
            self.eval(&args[2]).as_f64().unwrap_or(1.0) as u32
        } else {
            1
        };
        if row_num < 1 || col_num < 1 {
            return EvalValue::Error("#REF!".into());
        }
        let target = Coord::new(
            top_left.col + col_num - 1,
            top_left.row + row_num - 1,
        );
        self.resolve_cell(target)
    }

    /// MATCH(lookup_value, lookup_range, [match_type])
    fn eval_match(&self, args: &[Expr]) -> EvalValue {
        if args.len() < 2 {
            return EvalValue::Error("#VALUE!".into());
        }
        let lookup = self.eval(&args[0]);
        let (top_left, bottom_right) = match &args[1] {
            Expr::RangeRef { top_left, bottom_right } => (*top_left, *bottom_right),
            _ => return EvalValue::Error("#VALUE!".into()),
        };
        let match_type = if args.len() > 2 {
            self.eval(&args[2]).as_f64().unwrap_or(1.0) as i32
        } else {
            1
        };

        // Determine if horizontal or vertical range
        let is_row = top_left.row == bottom_right.row;
        let count = if is_row {
            (bottom_right.col - top_left.col + 1) as usize
        } else {
            (bottom_right.row - top_left.row + 1) as usize
        };

        let get_val = |i: usize| -> EvalValue {
            if is_row {
                self.resolve_cell(Coord::new(top_left.col + i as u32, top_left.row))
            } else {
                self.resolve_cell(Coord::new(top_left.col, top_left.row + i as u32))
            }
        };

        match match_type {
            0 => {
                // Exact match
                for i in 0..count {
                    if get_val(i) == lookup {
                        return EvalValue::Number((i + 1) as f64);
                    }
                }
                EvalValue::Error("#N/A".into())
            }
            1 => {
                // Largest value <= lookup (sorted ascending)
                let mut last_match = None;
                for i in 0..count {
                    let v = get_val(i);
                    match (v.as_f64(), lookup.as_f64()) {
                        (Some(a), Some(b)) if a <= b => last_match = Some(i),
                        (Some(a), Some(b)) if a > b => break,
                        _ => {}
                    }
                }
                match last_match {
                    Some(i) => EvalValue::Number((i + 1) as f64),
                    None => EvalValue::Error("#N/A".into()),
                }
            }
            -1 => {
                // Smallest value >= lookup (sorted descending)
                let mut last_match = None;
                for i in 0..count {
                    let v = get_val(i);
                    match (v.as_f64(), lookup.as_f64()) {
                        (Some(a), Some(b)) if a >= b => last_match = Some(i),
                        (Some(a), Some(b)) if a < b => break,
                        _ => {}
                    }
                }
                match last_match {
                    Some(i) => EvalValue::Number((i + 1) as f64),
                    None => EvalValue::Error("#N/A".into()),
                }
            }
            _ => EvalValue::Error("#VALUE!".into()),
        }
    }

//! Lexer for spreadsheet formulas.

#[cfg(not(feature = "std"))]
use alloc::{string::String, vec::Vec};
#[cfg(feature = "std")]
use std::vec::Vec;

#[derive(Debug, Clone, PartialEq)]
pub enum Token {
    Number(f64),
    StringLit(String),
    Ident(String),
    Colon,
    Comma,
    LParen,
    RParen,
    Plus,
    Minus,
    Star,
    Slash,
    Percent,
    Caret,
    Ampersand,
    Eq,
    NotEq,
    Lt,
    LtEq,
    Gt,
    GtEq,
}

pub struct Lexer<'a> {
    chars: core::str::CharIndices<'a>,
    peeked: Option<(usize, char)>,
}

impl<'a> Lexer<'a> {
    pub fn new(input: &'a str) -> Self {
        let input = input.trim().strip_prefix('=').unwrap_or(input.trim());
        let mut chars = input.char_indices();
        let peeked = chars.next();
        Self {
            chars,
            peeked,
        }
    }

    fn advance(&mut self) -> Option<(usize, char)> {
        let current = self.peeked;
        self.peeked = self.chars.next();
        current
    }

    fn peek(&self) -> Option<char> {
        self.peeked.map(|(_, c)| c)
    }

    fn skip_whitespace(&mut self) {
        while let Some(c) = self.peek() {
            if c.is_whitespace() {
                self.advance();
            } else {
                break;
            }
        }
    }

    pub fn next_token(&mut self) -> Option<Token> {
        self.skip_whitespace();
        let (_start_idx, ch) = self.advance()?;

        let token = match ch {
            '+' => Token::Plus,
            '-' => Token::Minus,
            '*' => Token::Star,
            '/' => Token::Slash,
            '%' => Token::Percent,
            '^' => Token::Caret,
            '&' => Token::Ampersand,
            ':' => Token::Colon,
            ',' => Token::Comma,
            '(' => Token::LParen,
            ')' => Token::RParen,
            '=' => Token::Eq,
            '<' => {
                if self.peek() == Some('>') {
                    self.advance();
                    Token::NotEq
                } else if self.peek() == Some('=') {
                    self.advance();
                    Token::LtEq
                } else {
                    Token::Lt
                }
            }
            '>' => {
                if self.peek() == Some('=') {
                    self.advance();
                    Token::GtEq
                } else {
                    Token::Gt
                }
            }
            '"' => {
                let mut s = String::new();
                while let Some((_, c)) = self.advance() {
                    if c == '"' {
                        break;
                    }
                    s.push(c);
                }
                Token::StringLit(s)
            }
            c if c.is_ascii_digit() || c == '.' => {
                let mut num_str = String::from(c);
                while let Some(peek_ch) = self.peek() {
                    if peek_ch.is_ascii_digit() || peek_ch == '.' {
                        num_str.push(peek_ch);
                        self.advance();
                    } else {
                        break;
                    }
                }
                let num = num_str.parse::<f64>().unwrap_or(0.0);
                Token::Number(num)
            }
            c if c.is_alphabetic() || c == '_' => {
                let mut ident = String::from(c);
                while let Some(peek_ch) = self.peek() {
                    if peek_ch.is_alphanumeric() || peek_ch == '_' || peek_ch == '$' {
                        ident.push(peek_ch);
                        self.advance();
                    } else {
                        break;
                    }
                }
                Token::Ident(ident)
            }
            _ => return None,
        };

        Some(token)
    }

    pub fn tokenize(mut self) -> Vec<Token> {
        let mut tokens = Vec::new();
        while let Some(tok) = self.next_token() {
            tokens.push(tok);
        }
        tokens
    }
}

//! Recursive-descent Pratt parser for spreadsheet formula expressions.

#[cfg(not(feature = "std"))]
use alloc::{boxed::Box, string::String, vec::Vec};

use super::{
    ast::{BinaryOp, Expr, UnaryOp},
    lexer::{Lexer, Token},
};
use crate::model::Coord;

#[derive(Debug, PartialEq)]
pub enum ParseError {
    UnexpectedToken(String),
    UnexpectedEof,
    InvalidReference(String),
}

pub struct Parser {
    tokens: Vec<Token>,
    pos: usize,
}

impl Parser {
    pub fn new(tokens: Vec<Token>) -> Self {
        Self { tokens, pos: 0 }
    }

    pub fn parse_str(input: &str) -> Result<Expr, ParseError> {
        let tokens = Lexer::new(input).tokenize();
        let mut parser = Self::new(tokens);
        let expr = parser.parse_expr(0)?;
        if !parser.is_eof() {
            return Err(ParseError::UnexpectedToken("Trailing tokens".into()));
        }
        Ok(expr)
    }

    fn peek(&self) -> Option<&Token> {
        self.tokens.get(self.pos)
    }

    fn advance(&mut self) -> Option<Token> {
        if self.pos < self.tokens.len() {
            let tok = self.tokens[self.pos].clone();
            self.pos += 1;
            Some(tok)
        } else {
            None
        }
    }

    fn is_eof(&self) -> bool {
        self.pos >= self.tokens.len()
    }

    fn parse_expr(&mut self, min_bp: u8) -> Result<Expr, ParseError> {
        let mut lhs = self.parse_primary()?;

        loop {
            let op = match self.peek() {
                Some(Token::Plus) => BinaryOp::Add,
                Some(Token::Minus) => BinaryOp::Sub,
                Some(Token::Star) => BinaryOp::Mul,
                Some(Token::Slash) => BinaryOp::Div,
                Some(Token::Percent) => BinaryOp::Mod,
                Some(Token::Caret) => BinaryOp::Pow,
                Some(Token::Ampersand) => BinaryOp::Concat,
                Some(Token::Eq) => BinaryOp::Eq,
                Some(Token::NotEq) => BinaryOp::NotEq,
                Some(Token::Lt) => BinaryOp::Lt,
                Some(Token::LtEq) => BinaryOp::LtEq,
                Some(Token::Gt) => BinaryOp::Gt,
                Some(Token::GtEq) => BinaryOp::GtEq,
                _ => break,
            };

            let (l_bp, r_bp) = infix_binding_power(op);
            if l_bp < min_bp {
                break;
            }

            self.advance(); // consume operator
            let rhs = self.parse_expr(r_bp)?;
            lhs = Expr::Binary {
                op,
                left: Box::new(lhs),
                right: Box::new(rhs),
            };
        }

        Ok(lhs)
    }

    fn parse_primary(&mut self) -> Result<Expr, ParseError> {
        let tok = self.advance().ok_or(ParseError::UnexpectedEof)?;

        match tok {
            Token::Number(n) => Ok(Expr::Number(n)),
            Token::StringLit(s) => Ok(Expr::Text(s)),
            Token::Minus => {
                let expr = self.parse_expr(10)?;
                Ok(Expr::Unary {
                    op: UnaryOp::Neg,
                    expr: Box::new(expr),
                })
            }
            Token::LParen => {
                let inner = self.parse_expr(0)?;
                match self.advance() {
                    Some(Token::RParen) => Ok(inner),
                    _ => Err(ParseError::UnexpectedToken("Expected ')'".into())),
                }
            }
            Token::Ident(name) => {
                let upper = name.to_ascii_uppercase();

                // Boolean literals
                if upper == "TRUE" {
                    return Ok(Expr::Boolean(true));
                }
                if upper == "FALSE" {
                    return Ok(Expr::Boolean(false));
                }

                // Function call: IDENT '(' args ')'
                if self.peek() == Some(&Token::LParen) {
                    self.advance(); // consume '('
                    let mut args = Vec::new();
                    if self.peek() != Some(&Token::RParen) {
                        loop {
                            args.push(self.parse_expr(0)?);
                            match self.peek() {
                                Some(Token::Comma) => {
                                    self.advance();
                                }
                                Some(Token::RParen) => break,
                                _ => return Err(ParseError::UnexpectedToken("Expected ',' or ')' in argument list".into())),
                            }
                        }
                    }
                    self.advance(); // consume ')'
                    return Ok(Expr::FunctionCall { name: upper, args });
                }

                // Range reference: IDENT ':' IDENT (e.g. A1:B10)
                if self.peek() == Some(&Token::Colon) {
                    self.advance(); // consume ':'
                    let right_tok = self.advance().ok_or(ParseError::UnexpectedEof)?;
                    if let Token::Ident(right_name) = right_tok {
                        let c1 = Coord::parse_a1(&strip_dollar(&name))
                            .ok_or_else(|| ParseError::InvalidReference(name.clone()))?;
                        let c2 = Coord::parse_a1(&strip_dollar(&right_name))
                            .ok_or_else(|| ParseError::InvalidReference(right_name.clone()))?;

                        let top_left = Coord::new(c1.col.min(c2.col), c1.row.min(c2.row));
                        let bottom_right = Coord::new(c1.col.max(c2.col), c1.row.max(c2.row));

                        return Ok(Expr::RangeRef { top_left, bottom_right });
                    } else {
                        return Err(ParseError::UnexpectedToken("Expected coordinate after ':'".into()));
                    }
                }

                // Single Cell Reference or Named Reference
                let cleaned = strip_dollar(&name);
                if let Some(coord) = Coord::parse_a1(&cleaned) {
                    Ok(Expr::CellRef(coord))
                } else {
                    Ok(Expr::NamedRef(name))
                }
            }
            _ => Err(ParseError::UnexpectedToken("Invalid primary expression".into())),
        }
    }
}

fn strip_dollar(s: &str) -> String {
    s.replace('$', "")
}

fn infix_binding_power(op: BinaryOp) -> (u8, u8) {
    match op {
        BinaryOp::Eq | BinaryOp::NotEq | BinaryOp::Lt | BinaryOp::LtEq | BinaryOp::Gt | BinaryOp::GtEq => (1, 2),
        BinaryOp::Concat => (3, 4),
        BinaryOp::Add | BinaryOp::Sub => (5, 6),
        BinaryOp::Mul | BinaryOp::Div | BinaryOp::Mod => (7, 8),
        BinaryOp::Pow => (10, 9), // Right-associative
    }
}

//! Embedded Documents, Cross-References, and Live Templates.
//!
//! As specified in Lotus 5-7-11 Section 6:
//! - Documents (Markdown notes, specifications, reports) are first-class workbook objects.
//! - Cell-to-document references allow cells to link to supporting context and section anchors (`doc#fragment`).
//! - Document-to-cell references embed live spreadsheet values (e.g. `{{cell:revenue}}` or `{{cell:A1}}`)
//!   or immutable point-in-time snapshots (`{{snapshot:A1}}`).

#[cfg(not(feature = "std"))]
use alloc::{format, string::String, string::ToString, vec, vec::Vec};

use crate::model::{Cell, Coord, Sheet, Workbook};

/// Representation of a parsed Markdown section heading.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct SectionHeading {
    pub level: usize,
    pub title: String,
    pub anchor: String,
    pub line_index: usize,
}

/// Convert a heading title into a URL/anchor slug:
/// - Lowercase ASCII
/// - Alphanumerics retained
/// - Spaces, underscores, slashes, and hyphens collapsed into single dashes
/// - Leading and trailing dashes trimmed
pub fn heading_to_slug(title: &str) -> String {
    let mut slug = String::new();
    let mut prev_dash = false;
    for ch in title.chars() {
        if ch.is_ascii_alphanumeric() {
            slug.push(ch.to_ascii_lowercase());
            prev_dash = false;
        } else if ch == ' ' || ch == '-' || ch == '_' || ch == '/' {
            if !prev_dash && !slug.is_empty() {
                slug.push('-');
                prev_dash = true;
            }
        }
    }
    if slug.ends_with('-') {
        slug.pop();
    }
    slug
}

/// Parse a line to check if it represents a Markdown heading (1 to 6 `#` characters followed by space/tab/EOL).
pub fn parse_heading_line(line: &str) -> Option<(usize, String)> {
    let trimmed = line.trim_start();
    if !trimmed.starts_with('#') {
        return None;
    }

    let mut level = 0;
    let chars: Vec<char> = trimmed.chars().collect();
    while level < chars.len() && chars[level] == '#' && level < 6 {
        level += 1;
    }

    if level == 0 {
        return None;
    }

    if level < chars.len() && chars[level] != ' ' && chars[level] != '\t' {
        return None;
    }

    let title: String = chars[level..].iter().collect();
    Some((level, title.trim().to_string()))
}

#[derive(Debug, Clone, PartialEq)]
pub struct Document {
    pub id: String,
    pub title: String,
    pub content_type: String, // e.g. "text/markdown"
    pub body: String,
}

impl Document {
    pub fn new(id: impl Into<String>, title: impl Into<String>, body: impl Into<String>) -> Self {
        Self {
            id: id.into(),
            title: title.into(),
            content_type: "text/markdown".into(),
            body: body.into(),
        }
    }

    /// Renders the document body by resolving live cell references:
    /// - `{{cell:A1}}` or `{{cell:name}}` resolves against the current live sheet state.
    pub fn render_live(&self, sheet: &Sheet, workbook: Option<&Workbook>) -> String {
        render_template_str(&self.body, sheet, workbook)
    }

    /// Return explicit `{{cell:target}}` references in document order.
    pub fn referenced_cells(&self) -> Vec<String> {
        let mut references = Vec::new();
        let mut remaining = self.body.as_str();
        while let Some(start) = remaining.find("{{") {
            remaining = &remaining[start + 2..];
            let Some(end) = remaining.find("}}") else { break };
            let tag = remaining[..end].trim();
            if let Some(target) = tag.strip_prefix("cell:") {
                let target = target.trim();
                if !target.is_empty() {
                    references.push(target.to_string());
                }
            }
            remaining = &remaining[end + 2..];
        }
        references
    }

    /// Whether this document explicitly references the supplied cell by A1 coordinate or name.
    pub fn references_cell(&self, cell: &Cell, sheet_id: &str, default_sheet_id: &str) -> bool {
        self.referenced_cells().iter().any(|target| {
            let (target_sheet_id, target_cell) = target.split_once('!')
                .unwrap_or((default_sheet_id, target.as_str()));
            target_sheet_id == sheet_id
                && (Coord::parse_a1(target_cell).is_some_and(|coord| coord == cell.coord)
                    || cell.name.as_deref() == Some(target_cell))
        })
    }

    /// Extract all Markdown section headings from the document body.
    pub fn headings(&self) -> Vec<SectionHeading> {
        let mut headings = Vec::new();
        for (line_index, line) in self.body.lines().enumerate() {
            if let Some((level, title)) = parse_heading_line(line) {
                let anchor = heading_to_slug(&title);
                headings.push(SectionHeading {
                    level,
                    title,
                    anchor,
                    line_index,
                });
            }
        }
        headings
    }

    /// Find a section heading matching a fragment (by canonical slug, case-insensitive anchor, or title).
    pub fn find_section(&self, fragment: &str) -> Option<SectionHeading> {
        let clean_frag = fragment.strip_prefix('#').unwrap_or(fragment).trim();
        if clean_frag.is_empty() {
            return None;
        }
        let frag_slug = heading_to_slug(clean_frag);

        self.headings().into_iter().find(|h| {
            h.anchor == clean_frag.to_ascii_lowercase()
                || h.anchor == frag_slug
                || h.title.eq_ignore_ascii_case(clean_frag)
        })
    }

    /// Check whether a section anchor/heading exists in this document.
    pub fn has_section(&self, fragment: &str) -> bool {
        self.find_section(fragment).is_some()
    }

    /// Extract the unrendered body text of a specific section (from the matching heading
    /// up to the next heading of equal or higher level, or end of document).
    pub fn section_content(&self, fragment: &str) -> Option<String> {
        let target = self.find_section(fragment)?;
        let lines: Vec<&str> = self.body.lines().collect();
        let mut section_lines = Vec::new();

        for (idx, line) in lines.iter().enumerate().skip(target.line_index) {
            if idx > target.line_index {
                if let Some((level, _)) = parse_heading_line(line) {
                    if level <= target.level {
                        break;
                    }
                }
            }
            section_lines.push(*line);
        }

        Some(section_lines.join("\n"))
    }

    /// Render a specific section of the document with live cell references resolved.
    pub fn render_section(&self, fragment: &str, sheet: &Sheet, workbook: Option<&Workbook>) -> Option<String> {
        let content = self.section_content(fragment)?;
        Some(render_template_str(&content, sheet, workbook))
    }
}

/// Render live cell templates in arbitrary text against the given sheet and workbook.
pub fn render_template_str(text: &str, sheet: &Sheet, workbook: Option<&Workbook>) -> String {
    let mut out = String::new();
    let chars: Vec<char> = text.chars().collect();
    let mut i = 0;

    while i < chars.len() {
        if i + 1 < chars.len() && chars[i] == '{' && chars[i + 1] == '{' {
            let start = i + 2;
            let mut end = start;
            while end + 1 < chars.len() && !(chars[end] == '}' && chars[end + 1] == '}') {
                end += 1;
            }

            if end + 1 < chars.len() {
                let tag: String = chars[start..end].iter().collect();
                let resolved = resolve_tag(tag.trim(), sheet, workbook);
                out.push_str(&resolved);
                i = end + 2;
                continue;
            }
        }
        out.push(chars[i]);
        i += 1;
    }

    out
}

fn resolve_tag(tag: &str, sheet: &Sheet, workbook: Option<&Workbook>) -> String {
    if let Some(target) = tag.strip_prefix("cell:") {
        let target = target.trim();
        let (target_sheet, target) = if let Some((sheet_id, cell_target)) = target.split_once('!') {
            let Some(target_sheet) = workbook.and_then(|workbook| workbook.get_sheet(sheet_id.trim())) else {
                return format!("<unknown-sheet:{}>", sheet_id.trim());
            };
            (target_sheet, cell_target.trim())
        } else {
            (sheet, target)
        };
        // Try coordinate first
        let coord = if let Some(c) = Coord::parse_a1(target) {
            Some(c)
        } else {
            // Try cell name
            target_sheet.cells.values().find(|c| c.name.as_deref() == Some(target)).map(|c| c.coord)
        };

        if let Some(c) = coord {
            if let Some(cell) = target_sheet.get_cell(c) {
                match &cell.value {
                    crate::model::CellValue::Number(n) => format!("{:.2}", n),
                    crate::model::CellValue::Text(t) => t.clone(),
                    crate::model::CellValue::Boolean(b) => (if *b { "TRUE" } else { "FALSE" }).into(),
                    crate::model::CellValue::Formula { cached_num, cached_text, cached_bool, expr } => {
                        if let Some(n) = cached_num {
                            format!("{:.2}", n)
                        } else if let Some(value) = cached_bool {
                            (if *value { "TRUE" } else { "FALSE" }).into()
                        } else if let Some(ref t) = cached_text {
                            t.clone()
                        } else {
                            format!("={}", expr)
                        }
                    }
                    _ => "<unsupported>".into(),
                }
            } else {
                "<empty>".into()
            }
        } else {
            format!("<unknown:{}>", target)
        }
    } else {
        format!("{{{{{}}}}}", tag)
    }
}

//! Z-Axis Historical Layers, Provenance, and Tombstone Overlays.
//!
//! As specified in Lotus 5-7-11 Section 3.6:
//! - The Z axis represents discrete historical layers in sequence (a stack of thin sheets).
//! - History is represented as overlays, avoiding duplicate full-workbook snapshots.
//! - Moving along the Z axis selects earlier or later historical states.
//! - Explicit tombstones track deletions so historical restoration remains unambiguous.

#[cfg(not(feature = "std"))]
use alloc::{collections::BTreeMap, format, string::String, vec::Vec};
#[cfg(feature = "std")]
use std::collections::BTreeMap;

use crate::model::{Cell, Coord, Sheet};

/// A mutation event recording historical provenance along the Z axis.
#[derive(Debug, Clone, PartialEq)]
pub struct HistoricalRecord {
    pub z: u32,
    pub coord: Coord,
    pub cell: Option<Cell>, // None represents an explicit Tombstone (deletion)
}

/// Manages multi-layered Z-axis history and provenance overlays.
#[derive(Debug, Clone, PartialEq)]
pub struct HistoryEngine {
    pub current_z: u32,
    pub max_z: u32,
    pub layers: BTreeMap<u32, Vec<HistoricalRecord>>,
}

impl HistoryEngine {
    pub fn new() -> Self {
        Self {
            current_z: 0,
            max_z: 0,
            layers: BTreeMap::new(),
        }
    }

    /// Record a snapshot layer of changed cells and deletions.
    pub fn record_layer(&mut self, changes: Vec<(Coord, Option<Cell>)>) -> Result<u32, &'static str> {
        let z = self.max_z.checked_add(1).ok_or("History Z axis is exhausted")?;
        self.insert_layer(z, changes)?;
        Ok(z)
    }

    /// Insert a serialized layer while preserving its Z value.
    /// Layers must arrive consecutively, and each cell may appear once per layer.
    pub fn insert_layer(
        &mut self,
        z: u32,
        changes: Vec<(Coord, Option<Cell>)>,
    ) -> Result<(), &'static str> {
        let expected_z = self.max_z.checked_add(1).ok_or("History Z axis is exhausted")?;
        if z != expected_z {
            return Err("History layers must have consecutive Z values starting at 1");
        }
        for (i, (coord, cell)) in changes.iter().enumerate() {
            if cell.as_ref().is_some_and(|value| value.coord != *coord) {
                return Err("History cell coordinate does not match its record coordinate");
            }
            if changes[..i].iter().any(|(prior, _)| prior == coord) {
                return Err("A cell may appear only once in a history layer");
            }
        }
        let records = changes
            .into_iter()
            .map(|(coord, cell)| HistoricalRecord { z, coord, cell })
            .collect();
        self.layers.insert(z, records);
        self.max_z = z;
        self.current_z = z;
        Ok(())
    }

    /// Construct a composite `Sheet` view at a specific discrete Z layer.
    ///
    /// Applies overlays sequentially from layer 0 up to `target_z`:
    /// - Unchanged cells persist.
    /// - Changed cells expose their state at or before `target_z`.
    /// - Deletions (tombstones) suppress cell visibility at and after that layer.
    pub fn composite_at_z(&self, base_sheet: &Sheet, target_z: u32) -> Sheet {
        let mut composite_sheet = Sheet::new(&base_sheet.id, format!("{} (Z={})", base_sheet.name, target_z));

        // Start with base sheet cells (layer 0)
        for (_coord, cell) in &base_sheet.cells {
            composite_sheet.set_cell(cell.clone());
        }

        // Apply layers 1..=target_z in chronological order
        for (z, records) in &self.layers {
            if *z > target_z {
                break;
            }
            for rec in records {
                match &rec.cell {
                    Some(updated_cell) => {
                        composite_sheet.set_cell(updated_cell.clone());
                    }
                    None => {
                        // Tombstone: explicitly remove cell from composite view
                        composite_sheet.cells.remove(&rec.coord);
                    }
                }
            }
        }

        composite_sheet
    }
}

//! Autonomous memory space management for OS-agnostic and bare-metal execution.
//!
//! Provides a self-contained bump and arena allocator operating over a contiguous
//! physical memory slice or pointer without relying on OS virtual memory or libc malloc.

#[cfg(not(feature = "std"))]
use alloc::vec::Vec;
use core::ptr::NonNull;

#[derive(Debug, PartialEq)]
pub enum MemoryError {
    OutOfMemory { requested: usize, available: usize },
    InvalidAlignment,
    NullPointer,
}

/// A contiguous, self-contained physical or virtual memory region.
/// Passed by an exokernel (like BareMetal OS), bootloader, or host substrate.
pub struct MemoryRegion {
    pub base: NonNull<u8>,
    pub size: usize,
}

impl MemoryRegion {
    /// Create a memory region from a raw pointer and size.
    ///
    /// # Safety
    /// The caller must ensure that `base` points to at least `size` bytes of valid, writable memory.
    pub unsafe fn from_raw(base: *mut u8, size: usize) -> Result<Self, MemoryError> {
        let ptr = NonNull::new(base).ok_or(MemoryError::NullPointer)?;
        Ok(Self { base: ptr, size })
    }

    /// Create a memory region borrowing a mutable byte slice.
    pub fn from_slice(slice: &mut [u8]) -> Self {
        let ptr = NonNull::new(slice.as_mut_ptr()).expect("Slice pointer cannot be null");
        Self {
            base: ptr,
            size: slice.len(),
        }
    }
}

/// A deterministic, self-contained Bump Allocator operating entirely within a `MemoryRegion`.
pub struct BumpArena {
    base: NonNull<u8>,
    capacity: usize,
    allocated: usize,
}

impl BumpArena {
    pub fn new(region: MemoryRegion) -> Self {
        Self {
            base: region.base,
            capacity: region.size,
            allocated: 0,
        }
    }

    /// Allocate `size` bytes aligned to `align` bytes.
    pub fn alloc(&mut self, size: usize, align: usize) -> Result<NonNull<u8>, MemoryError> {
        if !align.is_power_of_two() {
            return Err(MemoryError::InvalidAlignment);
        }

        let current_ptr = unsafe { self.base.as_ptr().add(self.allocated) as usize };
        let aligned_ptr = (current_ptr + align - 1) & !(align - 1);
        let padding = aligned_ptr - current_ptr;

        let total_needed = padding.checked_add(size).ok_or(MemoryError::OutOfMemory {
            requested: size,
            available: self.capacity.saturating_sub(self.allocated),
        })?;

        if self.allocated + total_needed > self.capacity {
            return Err(MemoryError::OutOfMemory {
                requested: size,
                available: self.capacity.saturating_sub(self.allocated),
            });
        }

        self.allocated += total_needed;
        NonNull::new(aligned_ptr as *mut u8).ok_or(MemoryError::NullPointer)
    }

    /// Reset the arena to zero allocations without releasing the underlying memory region.
    pub fn reset(&mut self) {
        self.allocated = 0;
    }

    pub fn used(&self) -> usize {
        self.allocated
    }

    pub fn available(&self) -> usize {
        self.capacity.saturating_sub(self.allocated)
    }
}

        let region = MemoryRegion::from_slice(&mut buffer);

        // Allocate 10 bytes with 8-byte alignment

//! Spreadsheet grid coordinate system (0-indexed internally, 1-indexed A1 externally).

#[cfg(not(feature = "std"))]
use alloc::string::String;
use core::fmt;

/// Spatial coordinate on a worksheet grid.
/// `col` is 0-indexed (0 = A, 1 = B, 26 = AA, etc.).
/// `row` is 0-indexed (0 = 1, 26 = 27, etc.).
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct Coord {
    pub col: u32,
    pub row: u32,
}

impl Coord {
    pub const fn new(col: u32, row: u32) -> Self {
        Self { col, row }
    }

    /// Parse an A1-notation reference (e.g. "A1", "D27", "AA100") into a `Coord`.
    pub fn parse_a1(s: &str) -> Option<Self> {
        let s = s.trim();
        if s.is_empty() {
            return None;
        }

        let mut col: u32 = 0;
        let mut row_start = 0;
        let mut has_letters = false;

        for (i, b) in s.bytes().enumerate() {
            if b.is_ascii_alphabetic() {
                has_letters = true;
                let val = (b.to_ascii_uppercase() - b'A' + 1) as u32;
                col = col.checked_mul(26)?.checked_add(val)?;
            } else if b.is_ascii_digit() {
                row_start = i;
                break;
            } else {
                return None;
            }
        }

        if !has_letters || row_start == 0 {
            return None;
        }

        let row_str = &s[row_start..];
        let row_1based: u32 = row_str.parse().ok()?;
        if row_1based == 0 {
            return None;
        }

        Some(Self {
            col: col - 1,
            row: row_1based - 1,
        })
    }

    /// Format this coordinate into standard A1-notation (e.g. `Coord { col: 3, row: 26 }` -> `"D27"`).
    pub fn to_a1(&self) -> String {
        let mut col_str = String::new();
        let mut c = self.col + 1;
        while c > 0 {
            let rem = ((c - 1) % 26) as u8;
            col_str.insert(0, (b'A' + rem) as char);
            c = (c - 1) / 26;
        }

        #[cfg(feature = "std")]
        return format!("{}{}", col_str, self.row + 1);

        #[cfg(not(feature = "std"))]
        {
            use alloc::format;
            format!("{}{}", col_str, self.row + 1)
        }
    }
}

impl fmt::Display for Coord {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{}", self.to_a1())
    }
}

#[cfg(test)]

    fn test_coord_a1_roundtrip() {
        let cases = [
            ("A1", Coord::new(0, 0)),
            ("B2", Coord::new(1, 1)),
            ("Z1", Coord::new(25, 0)),
            ("AA1", Coord::new(26, 0)),

        for (a1, coord) in cases {

//! Worksheet representation and cell collection.

#[cfg(not(feature = "std"))]
use alloc::{collections::BTreeMap, string::String, vec::Vec};
#[cfg(feature = "std")]
use std::collections::BTreeMap;

use super::{cell::Cell, coord::Coord};

#[derive(Debug, Clone, PartialEq)]
pub struct Sheet {
    pub id: String,
    pub name: String,
    pub cells: BTreeMap<Coord, Cell>,
}

impl Sheet {
    pub fn new(id: impl Into<String>, name: impl Into<String>) -> Self {
        Self {
            id: id.into(),
            name: name.into(),
            cells: BTreeMap::new(),
        }
    }

    /// Insert or update a cell in the sheet.
    pub fn set_cell(&mut self, cell: Cell) {
        self.cells.insert(cell.coord, cell);
    }

    /// Retrieve a reference to a cell at a specific coordinate.
    pub fn get_cell(&self, coord: Coord) -> Option<&Cell> {
        self.cells.get(&coord)
    }

    /// Retrieve a mutable reference to a cell at a specific coordinate.
    pub fn get_cell_mut(&mut self, coord: Coord) -> Option<&mut Cell> {
        self.cells.get_mut(&coord)
    }

    /// Calculate the furthest used column and row in this sheet.
    /// Returns `None` if the sheet has no cells.
    pub fn bounds(&self) -> Option<(u32, u32)> {
        if self.cells.is_empty() {
            return None;
        }

        let mut max_col = 0;
        let mut max_row = 0;
        for coord in self.cells.keys() {
            if coord.col > max_col {
                max_col = coord.col;
            }
            if coord.row > max_row {
                max_row = coord.row;
            }
        }

        Some((max_col, max_row))
    }
}

    pub extent_col: u32,
    pub extent_row: u32,
    pub entries: Vec<TocEntry>,

#[cfg(not(feature = "std"))]
use alloc::{collections::BTreeMap, format, string::String, string::ToString, vec, vec::Vec};
#[cfg(feature = "std")]
use std::collections::BTreeMap;

use super::{
    cell::{Cell, CellValue},
    coord::Coord,
    sheet::Sheet,
    toc::{TableOfContents, TocEntry},

use crate::{
    calc::ast::Expr,
    doc::{
        BrokenDocCellRef, BrokenDocRef, BrokenSectionRef, DanglingFormulaRef,
        Document, DocumentRefStatus, OrphanReport,
    },
    history::HistoryEngine,
};

#[derive(Debug, Clone, PartialEq)]
pub struct Workbook {
    pub id: String,
    pub title: String,
    pub calc_mode: String,
    pub sheets: Vec<Sheet>,
    pub documents: Vec<Document>,
    /// Historical cell overlays, keyed by worksheet ID. The base sheets are Z=0.
    pub history: BTreeMap<String, HistoryEngine>,
}

impl Workbook {
    pub fn new(id: impl Into<String>, title: impl Into<String>) -> Self {
        Self {
            id: id.into(),
            title: title.into(),
            calc_mode: "auto".into(),
            sheets: Vec::new(),
            documents: Vec::new(),
            history: BTreeMap::new(),
        }
    }

    pub fn with_document(mut self, doc: Document) -> Self {
        self.documents.push(doc);
        self
    }

    pub fn get_document(&self, id: &str) -> Option<&Document> {
        self.documents.iter().find(|d| d.id == id)
    }

    /// Documents explicitly linked from a cell, resolving each target's document ID.
    pub fn documents_linked_from_cell(&self, sheet_id: &str, coord: Coord) -> Vec<&Document> {
        self.documents_linked_from_cell_at_z(sheet_id, coord, 0)
    }

    pub fn documents_linked_from_cell_at_z(&self, sheet_id: &str, coord: Coord, z: u32) -> Vec<&Document> {
        let Some(sheet) = self.sheet_at_z(sheet_id, z) else { return Vec::new() };
        let Some(cell) = sheet.get_cell(coord) else {
            return Vec::new();
        };
        self.documents.iter().filter(|document| {
            cell.document_refs.iter().any(|target| document_ref_id(target) == document.id)
        }).collect()
    }

    /// Documents whose explicit live-cell templates refer to this cell by A1 coordinate or name.
    pub fn documents_referencing_cell(&self, sheet_id: &str, coord: Coord) -> Vec<&Document> {
        self.documents_referencing_cell_at_z(sheet_id, coord, 0)
    }

    pub fn documents_referencing_cell_at_z(&self, sheet_id: &str, coord: Coord, z: u32) -> Vec<&Document> {
        let Some(sheet) = self.sheet_at_z(sheet_id, z) else { return Vec::new() };
        let Some(cell) = sheet.get_cell(coord) else {
            return Vec::new();
        };
        let default_sheet_id = self.primary_sheet().map(|sheet| sheet.id.as_str()).unwrap_or("");
        self.documents.iter()
            .filter(|document| document.references_cell(cell, sheet_id, default_sheet_id))
            .collect()
    }

    /// Validate an outgoing document reference (e.g. "briefing" or "briefing#inputs"):
    /// - Checks whether the document ID exists.
    /// - If a section fragment is present, verifies that the document contains a matching heading.
    pub fn validate_document_ref(&self, target: &str) -> DocumentRefStatus {
        let (doc_id, fragment) = match target.split_once('#') {
            Some((id, frag)) => (id.trim(), Some(frag.trim())),
            None => (target.trim(), None),
        };
        let Some(doc) = self.get_document(doc_id) else {
            return DocumentRefStatus::UnresolvedDocument;
        };
        if let Some(frag) = fragment {
            if frag.is_empty() {
                DocumentRefStatus::Valid
            } else if let Some(heading) = doc.find_section(frag) {
                DocumentRefStatus::ValidWithSection(heading.title)
            } else {
                DocumentRefStatus::UnresolvedSection(frag.to_string())
            }
        } else {
            DocumentRefStatus::Valid
        }
    }

    /// Audit the workbook for orphaned entities and broken references across:
    /// 1. Outgoing cell document references pointing to missing documents or missing section anchors.
    /// 2. Document-embedded live cell templates (`{{cell:...}}`) pointing to missing sheets or missing cells.
    /// 3. Embedded documents with zero incoming cell references.
    /// 4. Formula cells with dangling dependencies on non-existent or empty cells.
    pub fn check_orphans(&self) -> OrphanReport {
        let mut report = OrphanReport::default();

        // 1. Broken cell -> document references
        for sheet in &self.sheets {
            for cell in sheet.cells.values() {
                for target in &cell.document_refs {
                    match self.validate_document_ref(target) {
                        DocumentRefStatus::UnresolvedDocument => {
                            let doc_id = target.split_once('#').map_or(target.as_str(), |(id, _)| id);
                            report.broken_document_refs.push(BrokenDocRef {
                                sheet_id: sheet.id.clone(),
                                coord: cell.coord,
                                cell_name: cell.name.clone(),
                                target: target.clone(),
                                reason: format!("Document '{}' not found", doc_id),
                            });
                        }
                        DocumentRefStatus::UnresolvedSection(frag) => {
                            let doc_id = target.split_once('#').map_or(target.as_str(), |(id, _)| id);
                            report.broken_section_refs.push(BrokenSectionRef {
                                sheet_id: sheet.id.clone(),
                                coord: cell.coord,
                                cell_name: cell.name.clone(),
                                document_id: doc_id.to_string(),
                                section_anchor: frag,
                                target: target.clone(),
                            });
                        }
                        _ => {}
                    }
                }
            }
        }

        // 2. Broken document -> cell references
        let default_sheet_id = self.primary_sheet().map(|s| s.id.as_str()).unwrap_or("");
        for doc in &self.documents {
            for target in doc.referenced_cells() {
                let (sheet_id, cell_ref) = target.split_once('!')
                    .unwrap_or((default_sheet_id, target.as_str()));
                let Some(sheet) = self.get_sheet(sheet_id) else {
                    report.broken_document_cell_refs.push(BrokenDocCellRef {
                        document_id: doc.id.clone(),
                        target: target.clone(),
                        reason: format!("Worksheet '{}' not found", sheet_id),
                    });
                    continue;
                };
                let found = if let Some(coord) = Coord::parse_a1(cell_ref) {
                    sheet.get_cell(coord).is_some()
                } else {
                    sheet.cells.values().any(|c| c.name.as_deref() == Some(cell_ref))
                };
                if !found {
                    report.broken_document_cell_refs.push(BrokenDocCellRef {
                        document_id: doc.id.clone(),
                        target: target.clone(),
                        reason: format!("Cell '{}' not found on sheet '{}'", cell_ref, sheet_id),
                    });
                }
            }
        }

        // 3. Unreferenced documents (documents not referenced by any cell's document_refs)
        for doc in &self.documents {
            let is_referenced = self.sheets.iter().any(|sheet| {
                sheet.cells.values().any(|cell| {
                    cell.document_refs.iter().any(|target| {
                        document_ref_id(target) == doc.id
                    })
                })
            });
            if !is_referenced {
                report.unreferenced_documents.push(doc.id.clone());
            }
        }

        // 4. Dangling formula dependencies
        for sheet in &self.sheets {
            for cell in sheet.cells.values() {
                if let CellValue::Formula { expr, .. } = &cell.value {
                    if let Ok(ast) = crate::calc::Parser::parse_str(expr) {
                        let mut raw_targets = Vec::new();
                        collect_raw_formula_targets(&ast, &mut raw_targets);
                        for raw_target in raw_targets {
                            match raw_target {
                                RawFormulaTarget::Coord(coord) => {
                                    if sheet.get_cell(coord).is_none() {
                                        report.dangling_formula_refs.push(DanglingFormulaRef {
                                            sheet_id: sheet.id.clone(),
                                            coord: cell.coord,
                                            formula_name: cell.name.clone(),
                                            target: format!("{}", coord),
                                        });
                                    }
                                }
                                RawFormulaTarget::Name(name) => {
                                    if !sheet.cells.values().any(|c| c.name.as_deref() == Some(&name)) {
                                        report.dangling_formula_refs.push(DanglingFormulaRef {
                                            sheet_id: sheet.id.clone(),
                                            coord: cell.coord,
                                            formula_name: cell.name.clone(),
                                            target: name,
                                        });
                                    }
                                }
                                RawFormulaTarget::Range(tl, br) => {
                                    let mut any_cell_found = false;
                                    for col in tl.col..=br.col {
                                        for row in tl.row..=br.row {
                                            if sheet.get_cell(Coord::new(col, row)).is_some() {
                                                any_cell_found = true;
                                                break;
                                            }
                                        }
                                        if any_cell_found {
                                            break;
                                        }
                                    }
                                    if !any_cell_found {
                                        report.dangling_formula_refs.push(DanglingFormulaRef {
                                            sheet_id: sheet.id.clone(),
                                            coord: cell.coord,
                                            formula_name: cell.name.clone(),
                                            target: format!("{}:{}", tl, br),
                                        });
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }

        report
    }

    pub fn with_sheet(mut self, sheet: Sheet) -> Self {
        self.sheets.push(sheet);
        self
    }

    pub fn primary_sheet(&self) -> Option<&Sheet> {
        self.sheets.first()
    }

    pub fn primary_sheet_mut(&mut self) -> Option<&mut Sheet> {
        self.sheets.first_mut()
    }

    pub fn get_sheet(&self, id: &str) -> Option<&Sheet> {
        self.sheets.iter().find(|s| s.id == id)
    }

    pub fn get_sheet_mut(&mut self, id: &str) -> Option<&mut Sheet> {
        self.sheets.iter_mut().find(|s| s.id == id)
    }

    pub fn history_for_sheet(&self, id: &str) -> Option<&HistoryEngine> {
        self.history.get(id)
    }

    /// Compose a worksheet at Z=0 (the base) or at a stored historical layer.
    pub fn sheet_at_z(&self, id: &str, z: u32) -> Option<Sheet> {
        let base = self.get_sheet(id)?;
        match self.history.get(id) {
            Some(history) if z <= history.max_z => Some(history.composite_at_z(base, z)),
            None if z == 0 => Some(base.clone()),
            _ => None,
        }
    }

    /// Record changes as a new historical layer without modifying the base sheet.
    pub fn record_history_layer(
        &mut self,
        sheet_id: &str,
        changes: Vec<(Coord, Option<Cell>)>,
    ) -> Result<u32, &'static str> {
        if self.get_sheet(sheet_id).is_none() {
            return Err("Sheet not found");
        }
        let history = self.history.entry(sheet_id.into()).or_insert_with(HistoryEngine::new);
        history.record_layer(changes)
    }

    /// Dynamically computes the Table of Contents (TOC) anchored at (maxX + 1, maxY + 1).
    pub fn compute_toc(&self) -> TableOfContents {
        let mut max_col = 0;
        let mut max_row = 0;
        let mut has_cells = false;
        let mut entries = Vec::new();

        for sheet in &self.sheets {
            if let Some((scol, srow)) = sheet.bounds() {
                has_cells = true;
                if scol > max_col {
                    max_col = scol;
                }
                if srow > max_row {
                    max_row = srow;
                }
            }

            for cell in sheet.cells.values() {
                if let Some(ref name) = cell.name {
                    let (type_name, byte_size) = match &cell.value {
                        CellValue::Blob { bytes } => ("blob", bytes.len()),
                        CellValue::Basic { source } => ("basic", source.len()),
                        CellValue::Formula { expr, .. } => ("formula", expr.len()),
                        CellValue::Text(t) => ("text", t.len()),
                        CellValue::Number(_) => ("number", 8),
                        CellValue::Boolean(_) => ("boolean", 1),
                        CellValue::Empty => ("empty", 0),
                        CellValue::Error(e) => ("error", e.len()),
                    };

                    entries.push(TocEntry {
                        sheet_id: sheet.id.clone(),
                        coord: cell.coord,
                        name: name.clone(),
                        type_name: String::from(type_name),
                        byte_size,
                    });
                }
            }
        }

        let (toc_col, toc_row) = if has_cells {
            (max_col + 1, max_row + 1)
        } else {
            (0, 0)
        };

        TableOfContents {
            pos: Coord::new(toc_col, toc_row),
            extent_col: max_col,
            extent_row: max_row,
            entries,
        }
    }

    /// Eject a named cell as an exact byte slice.
    pub fn eject_cell(&self, name: &str) -> Option<&[u8]> {
        for doc in &self.documents {
            if doc.id == name {
                return Some(doc.body.as_bytes());
            }
        }

        for sheet in &self.sheets {
            for cell in sheet.cells.values() {
                if cell.name.as_deref() == Some(name) {
                    return match &cell.value {
                        CellValue::Blob { bytes } => Some(bytes.as_slice()),
                        CellValue::Basic { source } => Some(source.as_bytes()),
                        CellValue::Text(t) => Some(t.as_bytes()),
                        _ => None,
                    };
                }
            }
        }
        None
    }

    /// Import a raw byte sequence into a named cell at the given coordinate.
    pub fn import_blob(
        &mut self,
        sheet_idx: usize,
        coord: Coord,
        name: impl Into<String>,
        bytes: Vec<u8>,
    ) -> Result<(), &'static str> {
        let sheet = self.sheets.get_mut(sheet_idx).ok_or("Sheet not found")?;
        sheet.set_cell(Cell::blob(coord, name, bytes));
        Ok(())
    }
}

enum RawFormulaTarget {
    Coord(Coord),
    Name(String),
    Range(Coord, Coord),
}

fn collect_raw_formula_targets(expr: &Expr, targets: &mut Vec<RawFormulaTarget>) {
    match expr {
        Expr::CellRef(c) => targets.push(RawFormulaTarget::Coord(*c)),
        Expr::NamedRef(name) => targets.push(RawFormulaTarget::Name(name.clone())),
        Expr::RangeRef { top_left, bottom_right } => {
            targets.push(RawFormulaTarget::Range(*top_left, *bottom_right));
        }
        Expr::Unary { expr, .. } => collect_raw_formula_targets(expr, targets),
        Expr::Binary { left, right, .. } => {
            collect_raw_formula_targets(left, targets);
            collect_raw_formula_targets(right, targets);
        }
        Expr::FunctionCall { args, .. } => {
            for arg in args {
                collect_raw_formula_targets(arg, targets);
            }
        }
        _ => {}
    }
}

fn document_ref_id(target: &str) -> &str {
    target.split_once('#').map_or(target, |(document_id, _)| document_id)
}

//! Deserialization of Lotus 5-7-11 workbooks from the uncompressed hybrid format.

#[cfg(not(feature = "std"))]
use alloc::{string::String, string::ToString, vec::Vec};

use crate::model::{
    cell::{Cell, CellValue},
    coord::Coord,
    sheet::Sheet,
    workbook::Workbook,
};

#[derive(Debug)]
pub enum ParseError {
    UnexpectedEof,
    InvalidTag(usize),
    MalformedAttribute(usize),
    MissingAttribute(&'static str),
    InvalidHistory(String),
    InvalidCoordinate(String),
    InvalidNumber(String),
    InvalidFormulaCache(String),
    BlobOverflow { expected: usize, available: usize },
}

pub fn deserialize_workbook(input: &[u8]) -> Result<Workbook, ParseError> {
    let mut scanner = Scanner::new(input);
    let mut workbook = Workbook::new("wb", "Lotus Workbook");
    let mut current_sheet: Option<Sheet> = None;
    let mut current_layer: Option<(String, u32, Vec<(Coord, Option<Cell>)>)> = None;
    let mut open_history_containers = 0usize;

    while !scanner.is_eof() {
        scanner.skip_whitespace_and_comments();
        if scanner.is_eof() {
            break;
        }

        if scanner.peek() == Some(b'<') {
            let tag = scanner.parse_tag_header()?;

            match tag.name {
                "workbook" => {
                    if !tag.is_closing {
                        if let Some(id) = tag.get_attr_unescaped("id") {
                            workbook.id = id;
                        }
                        if let Some(title) = tag.get_attr_unescaped("title") {
                            workbook.title = title;
                        }
                        if let Some(calc) = tag.get_attr("calculation") {
                            workbook.calc_mode = calc.to_string();
                        }
                    }
                }
                "toc" => {
                    // Skip over TOC contents during read, as TOC is calculated dynamically on demand
                    if !tag.is_self_closing && !tag.is_closing {
                        scanner.skip_until_closing_tag("toc")?;
                    }
                }
                "sheet" => {
                    if tag.is_closing {
                        if let Some(sheet) = current_sheet.take() {
                            workbook.sheets.push(sheet);
                        }
                    } else {
                        let id = tag.get_attr_unescaped("id").unwrap_or_else(|| "sheet".to_string());
                        let name = tag.get_attr_unescaped("name").unwrap_or_else(|| "Sheet".to_string());
                        current_sheet = Some(Sheet::new(id, name));
                    }
                }
                "documents" => {
                    // Container tag, scanner will process child document tags
                }
                "history" => {
                    if tag.is_closing {
                        open_history_containers = open_history_containers.checked_sub(1)
                            .ok_or_else(|| ParseError::InvalidHistory("Closing history without an open history block".into()))?;
                    } else if !tag.is_self_closing {
                        open_history_containers += 1;
                    }
                }
                "layer" => {
                    if tag.is_closing {
                        let (sheet_id, z, changes) = current_layer.take().ok_or_else(|| {
                            ParseError::InvalidHistory("Closing layer without an open layer".into())
                        })?;
                        if workbook.get_sheet(&sheet_id).is_none() {
                            return Err(ParseError::InvalidHistory(format!("Unknown history sheet '{}'", sheet_id)));
                        }
                        workbook.history.entry(sheet_id.clone()).or_insert_with(crate::history::HistoryEngine::new)
                            .insert_layer(z, changes)
                            .map_err(|message| ParseError::InvalidHistory(message.into()))?;
                    } else {
                        if current_layer.is_some() {
                            return Err(ParseError::InvalidHistory("Nested history layers are not allowed".into()));
                        }
                        let sheet_id = tag.get_attr_unescaped("sheet")
                            .ok_or(ParseError::MissingAttribute("sheet"))?;
                        let z = tag.get_attr("z").and_then(|value| value.parse::<u32>().ok())
                            .ok_or(ParseError::MissingAttribute("z"))?;
                        if tag.is_self_closing {
                            if workbook.get_sheet(&sheet_id).is_none() {
                                return Err(ParseError::InvalidHistory(format!("Unknown history sheet '{}'", sheet_id)));
                            }
                            workbook.history.entry(sheet_id.clone()).or_insert_with(crate::history::HistoryEngine::new)
                                .insert_layer(z, Vec::new())
                                .map_err(|message| ParseError::InvalidHistory(message.into()))?;
                        } else {
                            current_layer = Some((sheet_id, z, Vec::new()));
                        }
                    }
                }
                "delete" => {
                    if tag.is_closing || !tag.is_self_closing {
                        return Err(ParseError::InvalidHistory("History deletions must use self-closing <delete /> tags".into()));
                    }
                    let (_, _, changes) = current_layer.as_mut().ok_or_else(|| {
                        ParseError::InvalidHistory("Deletion outside a history layer".into())
                    })?;
                    let coord_str = tag.get_attr("ref").ok_or(ParseError::MissingAttribute("ref"))?;
                    let coord = Coord::parse_a1(coord_str)
                        .ok_or_else(|| ParseError::InvalidCoordinate(coord_str.to_string()))?;
                    changes.push((coord, None));
                }
                "document" => {
                    if !tag.is_closing {
                        let id = tag.get_attr_unescaped("id").unwrap_or_else(|| "doc".to_string());
                        let title = tag.get_attr_unescaped("title").unwrap_or_else(|| "Document".to_string());
                        let doc_type = tag.get_attr_unescaped("type").unwrap_or_else(|| "text/markdown".to_string());
                        let body = if let Some(len) = tag.get_attr("bytes").and_then(|value| value.parse::<usize>().ok()) {
                            let bytes = scanner.consume_exact_bytes(len)?;
                            scanner.expect_closing_tag("document")?;
                            core::str::from_utf8(bytes).unwrap_or("").to_string()
                        } else {
                            let inner_bytes = scanner.consume_until_closing_tag("document")?;
                            let mut s = core::str::from_utf8(inner_bytes).unwrap_or("").trim();
                            if s.starts_with("<![CDATA[") && s.ends_with("]]>") {
                                s = &s[9..s.len() - 3];
                                s = s.trim();
                            }
                            s.to_string()
                        };
                        let mut doc = crate::doc::Document::new(id, title, body);
                        doc.content_type = doc_type.to_string();
                        workbook.documents.push(doc);
                    }
                }
                "cell" => {
                    if tag.is_closing {
                        // Handled inline below
                    } else {
                        let coord_str = tag
                            .get_attr("ref")
                            .ok_or(ParseError::MissingAttribute("ref"))?;
                        let coord = Coord::parse_a1(coord_str)
                            .ok_or_else(|| ParseError::InvalidCoordinate(coord_str.to_string()))?;

                        let cell_type = tag.get_attr("type").unwrap_or("empty");
                        let cell_id = tag.get_attr_unescaped("id");
                        let cell_name = tag.get_attr_unescaped("name");

                        let value = if tag.is_self_closing {
                            CellValue::Empty
                        } else {
                            match cell_type {
                                "blob" => {
                                    let bytes_len: usize = tag
                                        .get_attr("bytes")
                                        .and_then(|s| s.parse().ok())
                                        .unwrap_or(0);

                                    let raw_slice = scanner.consume_exact_bytes(bytes_len)?;
                                    scanner.expect_closing_tag("cell")?;
                                    CellValue::Blob {
                                        bytes: raw_slice.to_vec(),
                                    }
                                }
                                "basic" => {
                                    let source = if let Some(len) = tag.get_attr("bytes").and_then(|value| value.parse::<usize>().ok()) {
                                        let bytes = scanner.consume_exact_bytes(len)?;
                                        scanner.expect_closing_tag("cell")?;
                                        core::str::from_utf8(bytes).unwrap_or("").to_string()
                                    } else {
                                        let inner_bytes = scanner.consume_until_closing_tag("cell")?;
                                        let mut s = core::str::from_utf8(inner_bytes).unwrap_or("").trim();
                                        if s.starts_with("<![CDATA[") && s.ends_with("]]>") {
                                            s = &s[9..s.len() - 3];
                                            s = s.trim();
                                        }
                                        s.to_string()
                                    };
                                    CellValue::Basic { source }
                                }
                                "formula" => {
                                    let expr = tag.get_attr_unescaped("formula").unwrap_or_default();
                                    let inner_bytes = scanner.consume_until_closing_tag("cell")?;
                                    let inner_str = core::str::from_utf8(inner_bytes).unwrap_or("");
                                    let cached_type = tag.get_attr("cachedType");
                                    let (cached_num, cached_text, cached_bool) = match cached_type {
                                        Some("number") => {
                                            let value = inner_str.trim().parse::<f64>()
                                                .map_err(|_| ParseError::InvalidNumber(inner_str.trim().to_string()))?;
                                            (Some(value), None, None)
                                        }
                                        Some("boolean") => (None, None, Some(inner_str.trim().eq_ignore_ascii_case("true"))),
                                        Some("text") => (None, Some(unescape_xml(inner_str)), None),
                                        Some("empty") | None => {
                                            let legacy = inner_str.trim();
                                            let number = legacy.parse::<f64>().ok();
                                            let text = if number.is_none() && !legacy.is_empty() { Some(unescape_xml(legacy)) } else { None };
                                            (number, text, None)
                                        }
                                        Some(other) => return Err(ParseError::InvalidFormulaCache(other.to_string())),
                                    };

                                    CellValue::Formula {
                                        expr,
                                        cached_num,
                                        cached_text,
                                        cached_bool,
                                    }
                                }
                                "number" => {
                                    let inner_bytes = scanner.consume_until_closing_tag("cell")?;
                                    let s = core::str::from_utf8(inner_bytes)
                                        .unwrap_or("")
                                        .trim();
                                    let n = s
                                        .parse::<f64>()
                                        .map_err(|_| ParseError::InvalidNumber(s.to_string()))?;
                                    CellValue::Number(n)
                                }
                                "text" => {
                                    let inner_bytes = scanner.consume_until_closing_tag("cell")?;
                                    let s = core::str::from_utf8(inner_bytes).unwrap_or("");
                                    CellValue::Text(unescape_xml(s))
                                }
                                "boolean" => {
                                    let inner_bytes = scanner.consume_until_closing_tag("cell")?;
                                    let s = core::str::from_utf8(inner_bytes)
                                        .unwrap_or("")
                                        .trim();
                                    CellValue::Boolean(s.eq_ignore_ascii_case("true"))
                                }
                                "error" => {
                                    let inner_bytes = scanner.consume_until_closing_tag("cell")?;
                                    let s = core::str::from_utf8(inner_bytes)
                                        .unwrap_or("")
                                        .trim();
                                    CellValue::Error(unescape_xml(s))
                                }
                                _ => {
                                    scanner.consume_until_closing_tag("cell")?;
                                    CellValue::Empty
                                }
                            }
                        };

                        let mut cell = Cell::new(coord, value);
                        cell.id = cell_id;
                        cell.name = cell_name;
                        cell.document_refs = tag.get_attr_unescaped("documentRefs")
                            .unwrap_or_default()
                            .split(',')
                            .map(str::trim)
                            .filter(|target| !target.is_empty())
                            .map(ToString::to_string)
                            .collect();
                        if matches!(&cell.value, CellValue::Basic { .. }) {
                            cell.basic_reads = tag.get_attr_unescaped("reads").unwrap_or_default()
                                .split(',').map(str::trim).filter(|name| !name.is_empty()).map(ToString::to_string).collect();
                            cell.basic_writes = tag.get_attr_unescaped("writes").unwrap_or_default()
                                .split(',').map(str::trim).filter(|name| !name.is_empty()).map(ToString::to_string).collect();
                        }

                        if let Some((_, _, changes)) = current_layer.as_mut() {
                            changes.push((coord, Some(cell)));
                        } else if let Some(ref mut sheet) = current_sheet {
                            sheet.set_cell(cell);
                        }
                    }
                }
                _ => {
                    // Ignore unknown tags safely
                    if !tag.is_self_closing && !tag.is_closing {
                        let _ = scanner.skip_until_closing_tag(tag.name);
                    }
                }
            }
        } else {
            scanner.advance(1);
        }
    }

    if current_layer.is_some() || open_history_containers != 0 {
        return Err(ParseError::UnexpectedEof);
    }
    if let Some(sheet) = current_sheet.take() {
        workbook.sheets.push(sheet);
    }

    Ok(workbook)

struct Tag<'a> {
    name: &'a str,
    is_closing: bool,
    is_self_closing: bool,
    attrs: Vec<(&'a str, &'a str)>,
}

impl<'a> Tag<'a> {
    /// Attribute value with XML entities decoded (for free-text attributes: id, name, title, formula).
    fn get_attr_unescaped(&self, key: &str) -> Option<String> {
        self.get_attr(key).map(unescape_xml)
    }

    fn get_attr(&self, key: &str) -> Option<&'a str> {
        self.attrs
            .iter()
            .find(|(k, _)| *k == key)
            .map(|(_, v)| *v)
    }
}

struct Scanner<'a> {
    data: &'a [u8],
    pos: usize,
}

impl<'a> Scanner<'a> {
    fn new(data: &'a [u8]) -> Self {
        Self { data, pos: 0 }
    }

    fn is_eof(&self) -> bool {
        self.pos >= self.data.len()
    }

    fn peek(&self) -> Option<u8> {
        self.data.get(self.pos).copied()
    }

    fn advance(&mut self, n: usize) {
        self.pos = (self.pos + n).min(self.data.len());
    }

    fn skip_whitespace_and_comments(&mut self) {
        loop {
            // Whitespace
            while self.pos < self.data.len() && self.data[self.pos].is_ascii_whitespace() {
                self.pos += 1;
            }

            // Comments
            if self.pos + 4 <= self.data.len() && &self.data[self.pos..self.pos + 4] == b"<!--" {
                self.pos += 4;
                while self.pos + 3 <= self.data.len() {
                    if &self.data[self.pos..self.pos + 3] == b"-->" {
                        self.pos += 3;
                        break;
                    }
                    self.pos += 1;
                }
            } else {
                break;
            }
        }
    }

    fn parse_tag_header(&mut self) -> Result<Tag<'a>, ParseError> {
        if self.peek() != Some(b'<') {
            return Err(ParseError::InvalidTag(self.pos));
        }
        self.advance(1); // skip '<'

        let is_closing = if self.peek() == Some(b'/') {
            self.advance(1);
            true
        } else {
            false
        };

        // Extract tag name
        let name_start = self.pos;
        while self.pos < self.data.len()
            && !self.data[self.pos].is_ascii_whitespace()
            && self.data[self.pos] != b'>'
            && self.data[self.pos] != b'/'
        {
            self.pos += 1;
        }

        let name = core::str::from_utf8(&self.data[name_start..self.pos])
            .map_err(|_| ParseError::InvalidTag(name_start))?;

        let mut attrs = Vec::new();
        let mut is_self_closing = false;

        // Parse attributes
        loop {
            while self.pos < self.data.len() && self.data[self.pos].is_ascii_whitespace() {
                self.pos += 1;
            }

            if self.pos >= self.data.len() {
                return Err(ParseError::UnexpectedEof);
            }

            if self.data[self.pos] == b'/' {
                is_self_closing = true;
                self.advance(1);
                continue;
            }

            if self.data[self.pos] == b'>' {
                self.advance(1); // end of tag
                break;
            }

            // Read attribute key
            let key_start = self.pos;
            while self.pos < self.data.len()
                && self.data[self.pos] != b'='
                && !self.data[self.pos].is_ascii_whitespace()
                && self.data[self.pos] != b'>'
                && self.data[self.pos] != b'/'
            {
                self.pos += 1;
            }

            let key = core::str::from_utf8(&self.data[key_start..self.pos])
                .map_err(|_| ParseError::MalformedAttribute(key_start))?;

            while self.pos < self.data.len() && self.data[self.pos].is_ascii_whitespace() {
                self.pos += 1;
            }

            if self.pos < self.data.len() && self.data[self.pos] == b'=' {
                self.advance(1);
                while self.pos < self.data.len() && self.data[self.pos].is_ascii_whitespace() {
                    self.pos += 1;
                }

                let quote = self.peek().ok_or(ParseError::UnexpectedEof)?;
                if quote == b'"' || quote == b'\'' {
                    self.advance(1);
                    let val_start = self.pos;
                    while self.pos < self.data.len() && self.data[self.pos] != quote {
                        self.pos += 1;
                    }
                    let val = core::str::from_utf8(&self.data[val_start..self.pos])
                        .map_err(|_| ParseError::MalformedAttribute(val_start))?;
                    self.advance(1); // consume closing quote
                    attrs.push((key, val));
                }
            }
        }

        Ok(Tag {
            name,
            is_closing,
            is_self_closing,
            attrs,
        })
    }

    fn consume_exact_bytes(&mut self, n: usize) -> Result<&'a [u8], ParseError> {
        if self.pos + n > self.data.len() {
            return Err(ParseError::BlobOverflow {
                expected: n,
                available: self.data.len().saturating_sub(self.pos),
            });
        }
        let slice = &self.data[self.pos..self.pos + n];
        self.pos += n;
        Ok(slice)
    }

    fn consume_until_closing_tag(&mut self, tag: &str) -> Result<&'a [u8], ParseError> {
        let needle = [b'<', b'/'];
        let start = self.pos;

        while self.pos + 2 + tag.len() + 1 <= self.data.len() {
            if &self.data[self.pos..self.pos + 2] == &needle {
                let tag_slice = &self.data[self.pos + 2..self.pos + 2 + tag.len()];
                if tag_slice.eq_ignore_ascii_case(tag.as_bytes()) {
                    let content = &self.data[start..self.pos];
                    self.pos += 2 + tag.len();
                    while self.pos < self.data.len() && self.data[self.pos] != b'>' {
                        self.pos += 1;
                    }
                    if self.pos < self.data.len() {
                        self.pos += 1; // skip '>'
                    }
                    return Ok(content);
                }
            }
            self.pos += 1;
        }

        Err(ParseError::UnexpectedEof)
    }

    fn expect_closing_tag(&mut self, tag: &str) -> Result<(), ParseError> {
        self.skip_whitespace_and_comments();
        if self.pos + 2 + tag.len() + 1 > self.data.len() {
            return Err(ParseError::UnexpectedEof);
        }
        if &self.data[self.pos..self.pos + 2] != b"</" {
            return Err(ParseError::InvalidTag(self.pos));
        }
        self.pos += 2;
        let tag_slice = &self.data[self.pos..self.pos + tag.len()];
        if !tag_slice.eq_ignore_ascii_case(tag.as_bytes()) {
            return Err(ParseError::InvalidTag(self.pos));
        }
        self.pos += tag.len();
        while self.pos < self.data.len() && self.data[self.pos] != b'>' {
            self.pos += 1;
        }
        if self.pos < self.data.len() {
            self.pos += 1;
        }
        Ok(())
    }

    fn skip_until_closing_tag(&mut self, tag: &str) -> Result<(), ParseError> {
        let _ = self.consume_until_closing_tag(tag)?;
        Ok(())
    }
}

fn unescape_xml(s: &str) -> String {
    // `&amp;` must be decoded last, otherwise "&amp;lt;" would collapse to "<".
    s.replace("&quot;", "\"")
        .replace("&lt;", "<")
        .replace("&gt;", ">")
        .replace("&amp;", "&")
}

//! Serialization of Lotus 5-7-11 workbooks to the uncompressed hybrid format.

#[cfg(not(feature = "std"))]
use alloc::{format, string::String, vec::Vec};

use crate::model::{Cell, CellValue, Workbook};

/// Serializes a `Workbook` into a `.lotus` byte buffer.
///
/// The resulting format is:
/// - Human-readable in Notepad.
/// - Clearly marked with ASCII tags in a hex editor.
/// - Directly inspectable and editable by an AI as structured text.
/// - Preserves raw, uncompressed binary payloads byte-for-byte in `<cell type="blob">` tags.
pub fn serialize_workbook(wb: &Workbook) -> Vec<u8> {
    let mut out = Vec::new();

    // 1. Header
    out.extend_from_slice(b"<!-- LOTUS-5-7-11 WORKBOOK v1.0 -->\n");
    let wb_tag = format!(
        "<workbook id=\"{}\" title=\"{}\" calculation=\"{}\">\n",
        escape_attr(&wb.id),
        escape_attr(&wb.title),
        escape_attr(&wb.calc_mode)
    );
    out.extend_from_slice(wb_tag.as_bytes());

    // 2. Dynamic Table of Contents (TOC)
    let toc = wb.compute_toc();
    let toc_tag = format!(
        "  <toc ref=\"{}\" extentX=\"{}\" extentY=\"{}\">\n",
        toc.pos.to_a1(),
        toc.extent_col,
        toc.extent_row
    );
    out.extend_from_slice(toc_tag.as_bytes());

    for entry in &toc.entries {
        let entry_line = format!(
            "    <entry sheet=\"{}\" ref=\"{}\" name=\"{}\" type=\"{}\" bytes=\"{}\" />\n",
            escape_attr(&entry.sheet_id),
            entry.coord.to_a1(),
            escape_attr(&entry.name),
            entry.type_name,
            entry.byte_size
        );
        out.extend_from_slice(entry_line.as_bytes());
    }
    out.extend_from_slice(b"  </toc>\n\n");

    // 3. Sheets and Cells
    for sheet in &wb.sheets {
        let sheet_header = format!(
            "  <sheet id=\"{}\" name=\"{}\">\n",
            escape_attr(&sheet.id),
            escape_attr(&sheet.name)
        );
        out.extend_from_slice(sheet_header.as_bytes());

        for cell in sheet.cells.values() {
            write_cell(&mut out, cell);
        }

        out.extend_from_slice(b"  </sheet>\n");
    }

    // 4. Embedded Documents
    if !wb.documents.is_empty() {
        out.extend_from_slice(b"\n  <documents>\n");
        for doc in &wb.documents {
            let doc_header = format!(
                "    <document id=\"{}\" title=\"{}\" type=\"{}\" bytes=\"{}\">",
                escape_attr(&doc.id),
                escape_attr(&doc.title),
                escape_attr(&doc.content_type),
                doc.body.as_bytes().len()
            );
            out.extend_from_slice(doc_header.as_bytes());
            out.extend_from_slice(doc.body.as_bytes());
            out.extend_from_slice(b"</document>\n");
        }
        out.extend_from_slice(b"  </documents>\n");
    }

    // 5. Historical Z-layer overlays. Base sheet content is Z=0.
    if !wb.history.is_empty() {
        out.extend_from_slice(b"\n  <history>\n");
        for (sheet_id, history) in &wb.history {
            for (z, records) in &history.layers {
                out.extend_from_slice(
                    format!("    <layer sheet=\"{}\" z=\"{}\">\n", escape_attr(sheet_id), z).as_bytes(),
                );
                for record in records {
                    if let Some(cell) = &record.cell {
                        write_cell(&mut out, cell);
                    } else {
                        out.extend_from_slice(
                            format!("    <delete ref=\"{}\" />\n", record.coord.to_a1()).as_bytes(),
                        );
                    }
                }
                out.extend_from_slice(b"    </layer>\n");
            }
        }
        out.extend_from_slice(b"  </history>\n");
    }

    // 6. Closing Tag
    out.extend_from_slice(b"</workbook>\n");

    out
}

fn write_cell(out: &mut Vec<u8>, cell: &Cell) {
    let mut attrs = format!(" ref=\"{}\"", cell.coord.to_a1());

    if let Some(ref id) = cell.id {
        attrs.push_str(&format!(" id=\"{}\"", escape_attr(id)));
    }

    if let Some(ref name) = cell.name {
        attrs.push_str(&format!(" name=\"{}\"", escape_attr(name)));
    }
    if !cell.document_refs.is_empty() {
        attrs.push_str(&format!(
            " documentRefs=\"{}\"",
            escape_attr(&cell.document_refs.join(","))
        ));
    }

    match &cell.value {
        CellValue::Empty => {
            let line = format!("    <cell{} type=\"empty\" />\n", attrs);
            out.extend_from_slice(line.as_bytes());
        }
        CellValue::Number(n) => {
            let line = format!("    <cell{} type=\"number\">{}</cell>\n", attrs, n);
            out.extend_from_slice(line.as_bytes());
        }
        CellValue::Text(t) => {
            let line = format!(
                "    <cell{} type=\"text\">{}</cell>\n",
                attrs,
                escape_xml_text(t)
            );
            out.extend_from_slice(line.as_bytes());
        }
        CellValue::Boolean(b) => {
            let line = format!("    <cell{} type=\"boolean\">{}</cell>\n", attrs, b);
            out.extend_from_slice(line.as_bytes());
        }
        CellValue::Formula {
            expr,
            cached_num,
            cached_text,
            cached_bool,
        } => {
            let mut val_str = String::new();
            let mut cached_type = String::new();
            if let Some(n) = cached_num {
                val_str = format!("{}", n);
                cached_type = "number".into();
            } else if let Some(value) = cached_bool {
                val_str = value.to_string();
                cached_type = "boolean".into();
            } else if let Some(ref t) = cached_text {
                val_str = escape_xml_text(t);
                cached_type = "text".into();
            }

            let cache_attr = if cached_type.is_empty() { String::new() } else {
                format!(" cachedType=\"{}\"", cached_type)
            };

            let line = format!(
                "    <cell{} type=\"formula\" formula=\"{}\"{}>{}</cell>\n",
                attrs,
                escape_attr(expr),
                cache_attr,
                val_str
            );
            out.extend_from_slice(line.as_bytes());
        }
        CellValue::Basic { source } => {
            if !cell.basic_reads.is_empty() {
                attrs.push_str(&format!(" reads=\"{}\"", escape_attr(&cell.basic_reads.join(","))));
            }
            if !cell.basic_writes.is_empty() {
                attrs.push_str(&format!(" writes=\"{}\"", escape_attr(&cell.basic_writes.join(","))));
            }
            let open = format!("    <cell{} type=\"basic\" bytes=\"{}\">", attrs, source.as_bytes().len());
            out.extend_from_slice(open.as_bytes());
            out.extend_from_slice(source.as_bytes());
            out.extend_from_slice(b"</cell>\n");
        }
        CellValue::Blob { bytes } => {
            let open = format!(
                "    <cell{} type=\"blob\" bytes=\"{}\">",
                attrs,
                bytes.len()
            );
            out.extend_from_slice(open.as_bytes());
            // Embed exact, uncompressed raw payload bytes
            out.extend_from_slice(bytes);
            out.extend_from_slice(b"</cell>\n");
        }
        CellValue::Error(e) => {
            let line = format!(
                "    <cell{} type=\"error\">{}</cell>\n",
                attrs,
                escape_xml_text(e)
            );
            out.extend_from_slice(line.as_bytes());
        }
    }
}

fn escape_attr(s: &str) -> String {
    let mut out = String::with_capacity(s.len());
    for c in s.chars() {
        match c {
            '"' => out.push_str("&quot;"),
            '&' => out.push_str("&amp;"),
            '<' => out.push_str("&lt;"),
            '>' => out.push_str("&gt;"),
            _ => out.push(c),
        }
    }
    out
}

fn escape_xml_text(s: &str) -> String {
    let mut out = String::with_capacity(s.len());
    for c in s.chars() {
        match c {
            '&' => out.push_str("&amp;"),
            '<' => out.push_str("&lt;"),
            '>' => out.push_str("&gt;"),
            _ => out.push(c),
        }
    }
    out
}

//! Shared Exokernel Substrate and Memory-Map Contract.
//!
//! Provides the memory window mapping and zero-copy shared mailbox interface
//! for running alongside `ggml` under BareMetal OS (Return Infinity) or custom unikernels.
//!
//! Layout in contiguous 64-bit physical address space:
//! ```text
//! +-------------------------------------------------------------------+
//! | Substrate Descriptor (Magic, ABI Version, Memory Boundaries)      |
//! +-------------------------------------------------------------------+
//! | Lotus 5-7-11 Memory Window (Workspace, DAG, BASIC, Arena)         |
//! +-------------------------------------------------------------------+
//! | Shared Parameter / Tensor Mailbox Window (Zero-Copy Transfer)    |
//! +-------------------------------------------------------------------+
//! | ggml Unikernel Memory Window (Model Weights, KV Cache, Compute)   |
//! +-------------------------------------------------------------------+
//! ```

#[cfg(not(feature = "std"))]
use alloc::vec::Vec;
use core::ptr::NonNull;

pub const SUBSTRATE_MAGIC: [u8; 8] = *b"LOTUSEXO";
pub const SUBSTRATE_VERSION: u16 = 1;

/// Memory region descriptor for an individual execution partition.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
#[repr(C)]
pub struct Partition {
    pub offset: usize,
    pub size: usize,
}

/// The Canonical Substrate Descriptor mapped at the base of the shared memory space.
#[derive(Debug, Clone, PartialEq, Eq)]
#[repr(C)]
pub struct SubstrateDescriptor {
    pub magic: [u8; 8],
    pub version: u16,
    pub flags: u16,
    pub total_memory: usize,
    pub lotus_partition: Partition,
    pub mailbox_partition: Partition,
    pub ggml_partition: Partition,
}

impl SubstrateDescriptor {
    /// Create a validated memory layout across total available physical memory.
    pub fn new_layout(total_memory: usize, lotus_size: usize, mailbox_size: usize) -> Result<Self, &'static str> {
        let desc_size = 64; // Reserved descriptor header size
        let needed = desc_size + lotus_size + mailbox_size;
        if total_memory < needed {
            return Err("Insufficient memory for substrate partitioning");
        }

        let lotus_partition = Partition {
            offset: desc_size,
            size: lotus_size,
        };

        let mailbox_partition = Partition {
            offset: desc_size + lotus_size,
            size: mailbox_size,
        };

        let ggml_offset = desc_size + lotus_size + mailbox_size;
        let ggml_size = total_memory - ggml_offset;

        let ggml_partition = Partition {
            offset: ggml_offset,
            size: ggml_size,
        };

        Ok(Self {
            magic: SUBSTRATE_MAGIC,
            version: SUBSTRATE_VERSION,
            flags: 0,
            total_memory,
            lotus_partition,
            mailbox_partition,
            ggml_partition,
        })
    }
}

/// Zero-copy Mailbox for direct tensor and parameter handoff between Lotus and ggml.
pub struct SharedMailbox {
    buffer: NonNull<u8>,
    size: usize,
}

impl SharedMailbox {
    pub unsafe fn from_raw(base: *mut u8, size: usize) -> Option<Self> {
        let ptr = NonNull::new(base)?;
        Some(Self { buffer: ptr, size })
    }

    pub fn write_payload(&mut self, offset: usize, data: &[u8]) -> Result<(), &'static str> {
        if offset + data.len() > self.size {
            return Err("Mailbox overflow");
        }
        unsafe {
            let dest = self.buffer.as_ptr().add(offset);
            core::ptr::copy_nonoverlapping(data.as_ptr(), dest, data.len());
        }
        Ok(())
    }

    pub fn read_payload<'a>(&self, offset: usize, len: usize) -> Result<&'a [u8], &'static str> {
        if offset + len > self.size {
            return Err("Mailbox underflow / read out of bounds");
        }
        unsafe {
            let src = self.buffer.as_ptr().add(offset);
            Ok(core::slice::from_raw_parts(src, len))
        }
    }
}

#[cfg(not(feature = "std"))]

impl WorkspaceSession {
    pub fn new() -> Self {
        Self {
            cursor: Coord::new(0, 0),
            edit_buffer: None,
            menu: MenuState::new(),
            status_message: "READY (Press '/' for Lotus Menu, Enter to edit cell)".into(),
            is_dirty: false,
        }
    }

    pub fn move_cursor(&mut self, delta_col: i32, delta_row: i32) {
        if self.edit_buffer.is_some() || self.menu.is_active {
            return;
        }

        let new_col = (self.cursor.col as i64 + i64::from(delta_col)).clamp(0, 16_383) as u32;
        let new_row = (self.cursor.row as i64 + i64::from(delta_row)).clamp(0, 1_048_575) as u32;
        self.cursor = Coord::new(new_col, new_row);
    }

    pub fn start_editing(&mut self, sheet: &Sheet) {
        if self.menu.is_active {
            return;
        }

        let current_text = match sheet.get_cell(self.cursor) {
            Some(cell) => match &cell.value {
                CellValue::Formula { expr, .. } => expr.clone(),
                CellValue::Number(n) => format!("{}", n),
                CellValue::Text(t) => t.clone(),
                CellValue::Boolean(b) => (if *b { "TRUE" } else { "FALSE" }).into(),
                _ => String::new(),
            },
            None => String::new(),
        };

        self.edit_buffer = Some(current_text);
        self.status_message = "EDIT (Enter to commit, Esc to cancel)".into();
    }

    pub fn cancel_editing(&mut self) {
        self.edit_buffer = None;
        self.status_message = "READY".into();
    }

    pub fn commit_editing(&mut self, sheet: &mut Sheet) {
        if let Some(text) = self.edit_buffer.take() {
            let text = text.trim();
            if text.is_empty() {
                sheet.cells.remove(&self.cursor);
            } else {
                let cell_val = if text.starts_with('=') {
                    CellValue::Formula {
                        expr: text.into(),
                        cached_num: None,
                        cached_text: None,
                        cached_bool: None,
                    }
                } else if let Ok(n) = text.parse::<f64>() {
                    CellValue::Number(n)
                } else if text.eq_ignore_ascii_case("true") {
                    CellValue::Boolean(true)
                } else if text.eq_ignore_ascii_case("false") {
                    CellValue::Boolean(false)
                } else {
                    CellValue::Text(text.into())
                };

                if let Some(cell) = sheet.cells.get_mut(&self.cursor) {
                    cell.value = cell_val;
                } else {
                    sheet.set_cell(Cell::new(self.cursor, cell_val));
                }
            }

            // Recalculate sheet
            let _ = CalcEngine::recalculate_sheet(sheet, None);
            self.status_message = format!("Cell {} updated.", self.cursor.to_a1());
        }
    }

    /// Renders the complete terminal workspace view including status bar,
    /// Lotus menu (or edit bar), and the active cell grid.
    pub fn render_workspace(&self, sheet: &Sheet, toc: &TableOfContents) -> String {
        self.render_workspace_with_viewport(sheet, toc, 6, 12)
    }

    /// Render the workspace using the supplied number of visible grid rows and columns.
    pub fn render_workspace_with_viewport(
        &self,
        sheet: &Sheet,
        toc: &TableOfContents,
        viewport_cols: u32,
        viewport_rows: u32,
    ) -> String {
        let mut screen = String::new();

        // Line 1: Cell Address & Current Content
        let cell_info = match sheet.get_cell(self.cursor) {
            Some(cell) => {
                let name = cell.name.as_deref().map(|n| format!(" ({})", n)).unwrap_or_default();
                let desc = match &cell.value {
                    CellValue::Formula { expr, cached_num, cached_text, cached_bool } => {
                        let cached = cached_num.map(|n| format!("{:.2}", n))
                            .or_else(|| cached_bool.map(|value| if value { "TRUE".into() } else { "FALSE".into() }))
                            .or_else(|| cached_text.clone()).unwrap_or_else(|| "<pending>".into());
                        format!("Formula: {} => {}", expr, cached)
                    }
                    CellValue::Number(n) => format!("Number: {:.2}", n),
                    CellValue::Text(t) => format!("Text: \"{}\"", t),
                    CellValue::Boolean(b) => format!("Boolean: {}", b),
                    CellValue::Basic { .. } => "Embedded BASIC Procedure".into(),
                    CellValue::Blob { bytes } => format!("Raw Binary Payload ({} bytes)", bytes.len()),
                    CellValue::Empty => "Empty".into(),
                    CellValue::Error(e) => format!("Error: {}", e),
                };
                let links = if !cell.document_refs.is_empty() {
                    format!(" [Links: {}]", cell.document_refs.join(", "))
                } else {
                    String::new()
                };
                format!("{}{}{}", desc, name, links)
            }
            None => "Empty".into(),
        };

        screen.push_str(&format!("{}{} | {:<6} {}\n", sheet.name, if self.is_dirty { "*" } else { "" }, self.cursor.to_a1(), cell_info));

        // Line 2 & Line 3: Menu Bar or Edit Buffer
        if self.menu.is_active {
            let (l1, l2) = self.menu.render_lines();
            screen.push_str(&format!("MENU: {}\n", l1));
            screen.push_str(&format!("      {}\n", l2));
        } else if self.edit_buffer.is_some() {
            screen.push_str(&format!(
                "EDIT: {} | Enter commits, Esc cancels\n\n",
                self.edit_buffer.as_deref().unwrap_or_default()
            ));
        } else {
            screen.push_str(&format!("STATUS: {}\n\n", self.status_message));
        }

        // Fixed grid window follows the cursor without drawing all cells from A1.
        screen.push_str(&crate::view::GridRenderer::render_viewport(
            sheet,
            Some(toc),
            self.cursor,
            viewport_cols,
            viewport_rows,
        ));

        screen
    }
}

#[derive(Debug, Clone, PartialEq)]
pub struct MenuItem {
    pub key: char,
    pub label: &'static str,
    pub description: &'static str,
    pub action: MenuAction,
    pub submenus: Vec<MenuItem>,
}

impl MenuItem {
    pub fn new(key: char, label: &'static str, description: &'static str, action: MenuAction) -> Self {
        Self {
            key: key.to_ascii_uppercase(),
            label,
            description,
            action,
            submenus: Vec::new(),
        }
    }

    pub fn with_submenus(mut self, submenus: Vec<MenuItem>) -> Self {
        self.submenus = submenus;
        self
    }
}

pub struct MenuState {
    pub is_active: bool,
    pub breadcrumbs: Vec<usize>, // Selected index at each menu depth
    pub root: Vec<MenuItem>,
}

impl MenuState {
    pub fn new() -> Self {
        Self {
            is_active: false,
            breadcrumbs: Vec::new(),
            root: default_lotus_menu(),
        }
    }

    /// Open or toggle the menu bar with the classic '/' key.
    pub fn open(&mut self) {
        self.is_active = true;
        self.breadcrumbs.clear();
        self.breadcrumbs.push(0);
    }

    pub fn close(&mut self) {
        self.is_active = false;
        self.breadcrumbs.clear();
    }

    /// Handle a key event when the menu is active.
    /// Returns an executed `MenuAction` if an action was triggered.
    pub fn handle_key(&mut self, key_char: Option<char>, is_left: bool, is_right: bool, is_enter: bool, is_esc: bool) -> MenuAction {
        if !self.is_active {
            return MenuAction::None;
        }

        // Escape: Go up one level or dismiss menu
        if is_esc {
            if self.breadcrumbs.len() > 1 {
                self.breadcrumbs.pop();
            } else {
                self.close();
            }
            return MenuAction::None;
        }

        let current_items = self.current_level_items();
        let current_idx = *self.breadcrumbs.last().unwrap_or(&0);

        if is_left {
            let next_idx = if current_idx == 0 {
                current_items.len().saturating_sub(1)
            } else {
                current_idx - 1
            };
            if let Some(last) = self.breadcrumbs.last_mut() {
                *last = next_idx;
            }
            return MenuAction::None;
        }

        if is_right {
            let next_idx = (current_idx + 1) % current_items.len().max(1);
            if let Some(last) = self.breadcrumbs.last_mut() {
                *last = next_idx;
            }
            return MenuAction::None;
        }

        // First-letter shortcut jump (e.g. 'F' for File, 'S' for Save)
        if let Some(c) = key_char {
            let upper = c.to_ascii_uppercase();
            if let Some(pos) = current_items.iter().position(|item| item.key == upper) {
                if let Some(last) = self.breadcrumbs.last_mut() {
                    *last = pos;
                }
                return self.activate_current();
            }
        }

        if is_enter {
            return self.activate_current();
        }

        MenuAction::None
    }

    fn activate_current(&mut self) -> MenuAction {
        let current_idx = *self.breadcrumbs.last().unwrap_or(&0);
        let items = self.current_level_items();

        if let Some(selected) = items.get(current_idx) {
            if !selected.submenus.is_empty() {
                self.breadcrumbs.push(0);
                MenuAction::None
            } else {
                let action = selected.action.clone();
                if action != MenuAction::None {
                    self.close();
                }
                action
            }
        } else {
            MenuAction::None
        }
    }

    pub fn current_level_items(&self) -> &[MenuItem] {
        if self.breadcrumbs.is_empty() {
            return &self.root;
        }

        let mut current = &self.root;
        for &idx in &self.breadcrumbs[..self.breadcrumbs.len() - 1] {
            if let Some(item) = current.get(idx) {
                current = &item.submenus;
            } else {
                return &[];
            }
        }
        current
    }

    /// Returns (Line 1: Menu Items string, Line 2: Explanation/Submenu string).
    pub fn render_lines(&self) -> (String, String) {
        if !self.is_active {
            return (String::new(), String::new());
        }

        let items = self.current_level_items();
        let selected_idx = *self.breadcrumbs.last().unwrap_or(&0);

        let mut line1 = String::new();
        for (i, item) in items.iter().enumerate() {
            if i == selected_idx {
                line1.push_str(&format!("[{}] ", item.label));
            } else {
                line1.push_str(&format!(" {}  ", item.label));
            }
        }

        let line2 = if let Some(selected) = items.get(selected_idx) {
            if !selected.submenus.is_empty() {
                let mut sub_preview = String::new();
                for sub in &selected.submenus {
                    sub_preview.push_str(&format!("{} ", sub.label));
                }
                format!("{}: {}", selected.description, sub_preview)
            } else {
                selected.description.into()
            }
        } else {
            String::new()
        };

        (line1, line2)
    }
}

//! Terminal grid rasterizer and visual spreadsheet formatter.

#[cfg(not(feature = "std"))]
use alloc::{format, string::String, vec, vec::Vec};

use crate::model::{CellValue, Coord, Sheet, TableOfContents};

pub struct GridRenderer;

impl GridRenderer {
    /// Render a fixed-size window around the active cell.
    pub fn render_viewport(
        sheet: &Sheet,
        toc: Option<&TableOfContents>,
        cursor: Coord,
        viewport_cols: u32,
        viewport_rows: u32,
    ) -> String {
        let cols = viewport_cols.max(1);
        let rows = viewport_rows.max(1);
        let start_col = cursor.col.saturating_sub(cols / 2);
        let start_row = cursor.row.saturating_sub(rows / 2);
        let row_header_width = format!("{}", start_row.saturating_add(rows)).len().max(3) + 1;
        // Fixed-width cells keep the viewport within the terminal width; long values
        // are truncated in-grid while the status line shows the selected cell in full.
        let inner_widths = vec![10usize; cols as usize];

        let mut out = String::new();
        out.push_str(&format!("{:>width$} │", "", width = row_header_width));
        for offset in 0..cols {
            let col = start_col.saturating_add(offset);
            let label: String = Coord::new(col, 0).to_a1().chars().filter(|c| c.is_alphabetic()).collect();
            let width = inner_widths[offset as usize] + 2;
            out.push_str(&format!(" {:^width$} │", label, width = width));
        }
        out.push('\n');
        out.push_str(&format!("{:─>width$}─┼", "", width = row_header_width));
        for width in &inner_widths {
            out.push_str(&format!("{:─>width$}─┼", "", width = width + 2));
        }
        out.push('\n');

        for row_offset in 0..rows {
            let row = start_row.saturating_add(row_offset);
            let marker = if row == cursor.row { ">" } else { " " };
            let row_label = format!("{}{}", marker, row + 1);
            out.push_str(&format!("{:>width$} │", row_label, width = row_header_width));

            for col_offset in 0..cols {
                let col = start_col.saturating_add(col_offset);
                let coord = Coord::new(col, row);
                let inner_width = inner_widths[col_offset as usize];
                let width = inner_width + 2;
                let is_toc = toc.is_some_and(|table| table.pos == coord);
                if is_toc {
                    out.push_str(&format!(" {:^width$} │", "«TOC»", width = width));
                } else if let Some(cell) = sheet.get_cell(coord) {
                    let display = fit_cell(&cell_display_string(&cell.value), inner_width);
                    if coord == cursor {
                        out.push_str(&format!(" [{:<width$}] │", display, width = inner_width));
                    } else {
                        out.push_str(&format!(" {:<width$} │", display, width = width));
                    }
                } else if coord == cursor {
                    out.push_str(&format!(" [{:width$}] │", "", width = inner_width));
                } else {
                    out.push_str(&format!(" {:width$} │", "", width = width));
                }
            }
            out.push('\n');
        }
        out
    }

    /// Renders a worksheet as a formatted ASCII/Unicode terminal grid.
    pub fn render_sheet(sheet: &Sheet, toc: Option<&TableOfContents>, max_display_cols: Option<u32>, max_display_rows: Option<u32>) -> String {
        let (actual_max_col, actual_max_row) = sheet.bounds().unwrap_or((0, 0));
        let toc_col = toc.map(|t| t.pos.col).unwrap_or(0);
        let toc_row = toc.map(|t| t.pos.row).unwrap_or(0);

        let max_col = max_display_cols
            .unwrap_or(actual_max_col.max(4))
            .max(actual_max_col)
            .max(toc_col);
        let max_row = max_display_rows
            .unwrap_or(actual_max_row.max(9))
            .max(actual_max_row)
            .max(toc_row);

        // Determine column widths (default 12 chars)
        let mut col_widths: Vec<usize> = vec![12; (max_col + 1) as usize];

        // Header width for row numbers
        let row_num_width = format!("{}", max_row + 1).len().max(3);

        // Check if any cell content requires wider columns
        for cell in sheet.cells.values() {
            let col = cell.coord.col as usize;
            if col < col_widths.len() {
                let text = cell_display_string(&cell.value);
                if text.len() + 2 > col_widths[col] {
                    col_widths[col] = (text.len() + 2).min(30);
                }
            }
        }

        let mut out = String::new();

        // 1. Column Headers (A, B, C...)
        out.push_str(&format!("{:>width$} │", "", width = row_num_width));
        for col in 0..=max_col {
            let col_name = Coord::new(col, 0).to_a1();
            let col_letter: String = col_name.chars().filter(|c| c.is_alphabetic()).collect();
            let w = col_widths[col as usize];
            out.push_str(&format!(" {:^width$} │", col_letter, width = w));
        }
        out.push('\n');

        // Divider
        out.push_str(&format!("{:─>width$}─┼", "", width = row_num_width));
        for col in 0..=max_col {
            let w = col_widths[col as usize];
            out.push_str(&format!("{:─>width$}─┼", "", width = w));
        }
        out.push('\n');

        // 2. Rows
        for row in 0..=max_row {
            out.push_str(&format!("{:>width$} │", row + 1, width = row_num_width));

            for col in 0..=max_col {
                let coord = Coord::new(col, row);
                let w = col_widths[col as usize];

                let is_toc = toc.map(|t| t.pos == coord).unwrap_or(false);

                if is_toc {
                    out.push_str(&format!(" {:^width$} │", "«TOC»", width = w));
                } else if let Some(cell) = sheet.get_cell(coord) {
                    let display = cell_display_string(&cell.value);
                    let is_numeric = match &cell.value {
                        CellValue::Number(_) => true,
                        CellValue::Formula { cached_num: Some(_), .. } => true,
                        _ => false,
                    };

                    if is_numeric {
                        out.push_str(&format!(" {:>width$} │", display, width = w));
                    } else {
                        out.push_str(&format!(" {:<width$} │", display, width = w));
                    }
                } else {
                    out.push_str(&format!(" {:width$} │", "", width = w));
                }
            }
            out.push('\n');
        }

        out
    }
}

fn fit_cell(value: &str, width: usize) -> String {
    let char_count = value.chars().count();
    if char_count <= width {
        return value.to_string();
    }
    if width == 0 {
        return String::new();
    }
    let mut fitted: String = value.chars().take(width - 1).collect();
    fitted.push('…');
    fitted
}

fn cell_display_string(val: &CellValue) -> String {
    match val {
        CellValue::Empty => String::new(),
        CellValue::Number(n) => format!("{:.2}", n),
        CellValue::Text(t) => t.clone(),
        CellValue::Boolean(b) => (if *b { "TRUE" } else { "FALSE" }).into(),
        CellValue::Formula { cached_num, cached_text, cached_bool, expr } => {
            if let Some(n) = cached_num {
                format!("{:.2}", n)
            } else if let Some(value) = cached_bool {
                (if *value { "TRUE" } else { "FALSE" }).into()
            } else if let Some(ref t) = cached_text {
                t.clone()
            } else {
                format!("={}", expr)
            }
        }
        CellValue::Basic { .. } => "[BASIC]".into(),
        CellValue::Blob { bytes } => format!("[BLOB:{}B]", bytes.len()),
        CellValue::Error(e) => e.clone(),
    }
}
